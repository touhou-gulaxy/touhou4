touhou_psionic_echo_sanae = {
    category = minor_patron
    color = { 0.647 0.052 0.858 1 }
    text_color_code = "U"
    first_contact_event = toho.8200

    pre_communications_name_format = "{$core_name_touhou_psionic_echo_sanae$}"

    potential = {
        OR = {
            has_authority = auth_gensokyo
            has_country_flag = touhou_mod_country
        }
        has_country_flag = toho_select_ruler_sanae
        ruler = {
            is_touhou_special_leader = yes
            has_trait = leader_trait_touhou_psionic
            has_trait = leader_trait_sanae
        }
    }

    position = { x = 0.25 y = -0.25 }

    passive_accord = {
        name = touhou_psionic_echo_sanae
        hide_desc = yes
        triggered_modifier = {
            megastructure_build_speed_add = 0.05
            megastructures_upkeep_mult = -0.25
            megastructures_produces_mult = 0.25
        }
        on_unlock = {
            custom_tooltip = touhou_psionic_echo_sanae_passive_accord_effect
            set_country_flag = touhou_psionic_echo_sanae_passive_accord_unlocked
        }
    }
}
