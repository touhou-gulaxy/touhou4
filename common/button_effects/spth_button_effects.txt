spth_settings_refresh_data = {
    potential = { always = yes }
    allow = { hidden_trigger = { from = { NOT = { has_country_flag = refreshing_spth_data_cooldown } } } }
    effect = {
        custom_tooltip = gensokyo_start.1145.refresh_data.tooltip
        hidden_effect = {
            from = {
                set_country_flag = refreshing_spth_data_cooldown
                set_update_modifiers_batch = begin
                every_owned_planet = {
                    limit = {
                        OR = {
                            has_modifier = spth_planet_pop_convert
                            has_deposit = d_hakurei_barrier
                            NOT = { is_variable_set = planet_weak_genso_job_produce_faith_modifier }
                        }
                    }
                    if = {
                        limit = { has_deposit = d_hakurei_barrier }
                        spth_effect_refresh_barrier_and_faith = yes
                    }
                    if = {
                        limit = { NOT = { is_variable_set = planet_weak_genso_job_produce_faith_modifier } }
                        set_variable = { which = planet_weak_genso_job_produce_faith_modifier value = 1 }
                    }
                    spth_count_to_be_converted_pops = yes
                    if = {
                        limit = { has_modifier = spth_planet_pop_convert }
                        remove_modifier = spth_planet_pop_convert
                        add_modifier = {
                            modifier = spth_planet_pop_convert
                            multiplier = spth_to_be_converted_pops_counter
                        }
                    }
                }
                set_update_modifiers_batch = end
                remove_country_flag = refreshing_spth_data_cooldown
            }
        }
    }
}

        hidden_trigger = { is_colony = yes NOT = { has_planet_flag = touhou_move_in_pop_lock } }
        has_deposit = d_hakurei_barrier
        custom_tooltip = {
            fail_text = touhou_move_in_touhou_move_in_pop_add_no_pop
            is_variable_set = touhou_move_in_pop_steps
            count_owned_pop_amount = {
                limit = { is_same_species = owner_main_species }
                count > touhou_move_in_pop_steps
            }
        }
    }
    effect = {
        custom_tooltip = district_gensokyo_barrier_move_in_pop_plus_tt
        hidden_effect = {
            set_planet_flag = touhou_move_in_pop_lock
            set_variable = { which = touhou_move_in_pop_count value = touhou_move_in_pop_steps }
            every_owned_pop_group = {
                limit = { is_same_species = owner_main_species }
                if = {
                    limit = { prev = { check_variable = { which = touhou_move_in_pop_count value > 0 } } }
                    set_variable = { which = touhou_move_in_pop_group_pop_count value = trigger:pop_group_size }
                    if = {
                        limit = { check_variable = { which = touhou_move_in_pop_group_pop_count value >= prev.touhou_move_in_pop_count } }
                        clear_variable = touhou_move_in_pop_group_pop_count
                        remove_pop_amount = prev.touhou_move_in_pop_count
                        prev = { set_variable = { which = touhou_move_in_pop_count value = 0 } }
                    } else = {
                        prev = { subtract_variable = { which = touhou_move_in_pop_count value = prev.touhou_move_in_pop_group_pop_count } }
                        kill_all_pop = yes
                    }
                }
            }
            if = {
                limit = { check_variable = { which = touhou_move_in_pop_count value < 0 } }
                multiply_variable = { which = touhou_move_in_pop_count value = -1 }
                if = {
                    limit = { any_owned_pop_group = { is_same_species = owner_main_species } }
                    random_owned_pop_group = {
                        limit = { is_same_species = owner_main_species }
                        add_pop_amount = prev.touhou_move_in_pop_count
                    }
                } else = {
                    create_pop_group = { species = owner_main_species size = prev.touhou_move_in_pop_count }
                }
            }
            owner = {
                if = {
                    limit = { NOT = { is_variable_set = touhou_move_in_pop_amount } }
                    set_variable = { which = touhou_move_in_pop_amount value = 0 }
                }
                change_variable = { which = touhou_move_in_pop_amount value = prev.touhou_move_in_pop_steps }
            }
            remove_planet_flag = touhou_move_in_pop_lock
        }
    }
}
