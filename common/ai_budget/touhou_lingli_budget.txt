sr_lingli_expenditure_ships = {
    resource = sr_lingli
    type = expenditure
    category = ships

    potential = {
        is_touhou_player_faction = yes
    }

    weight = {
        weight = 500
        # spend more sr_lingli building ships while at war
        modifier = {
            factor = 2
            OR = {
                recently_lost_war = yes
                is_at_war = yes
            }
        }
        # spend less sr_lingli on ships if over capacity
        modifier = {
            factor = 0.25
            used_naval_capacity_percent >= 1.0
        }
    }

    desired_min = {
        base = 0
		modifier = {
			add = 100
		}
        modifier = {
            factor = trigger:years_passed
        }
    }
}
