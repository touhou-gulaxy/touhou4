hidden_effect = {
    if = {
        limit = { is_touhou_ship = yes }
        if = {
            limit = { NOT = { is_variable_set = touhou_steal_modifier_on_hit_dynamic_threshold_$modifier$ } }
            set_variable = {
                which = touhou_steal_modifier_on_hit_dynamic_threshold_$modifier$
                value = value:touhou_steal_modifier_on_hit_dynamic_threshold|factor|$threshold$|min_value|0|
            }
        }
        if = {
            limit = { check_modifier_value = { modifier = $modifier$ value > touhou_steal_modifier_on_hit_dynamic_threshold_$modifier$ } }
            touhou_steal_modifier_effect = {
                modifier = $modifier$
                value = $value$
                duration = $duration$
            }
        }
    } else = {
        if = {
            limit = { check_modifier_value = { modifier = $modifier$ value > $threshold$ } }
            touhou_steal_modifier_effect = {
                modifier = $modifier$
                value = $value$
                duration = $duration$
            }
        }
    }
}
