if = {
    limit = {
        exists = from from = { NOT = { has_ship_flag = touhou_aoe_damage_template_$size$_$type$ } }
    }
    from = { set_timed_ship_flag = { flag = touhou_aoe_damage_template_$size$_$type$ days = $cooldown$ } }
    set_variable = {
        which = touhou_aoe_damage_template_$size$_$type$_total_self
        value = from.value:touhou_aoe_damage_template_hit_damage|base_damage|$base_damage$|size_tag|$size_tag$|bonus_factor|$bonus_factor$|anti_psionic|$anti_psionic$|archaeotech|$archaeotech$|artillery|$artillery$|difficulty_scale|$difficulty_scale$|difficulty_scale_player|$difficulty_scale_player$|energy|$energy$|explosive|$explosive$|gravity|$gravity$|gungnir|$gungnir$|kappa|$kappa$|kinetic|$kinetic$|knife|$knife$|laevatain|$laevatain$|lingli|$lingli$|loukanken|$loukanken$|lunar_capital|$lunar_capital$|lunar_capital_2|$lunar_capital_2$|needle|$needle$|point_defense|$point_defense$|psionic|$psionic$|reimu|$reimu$|resurrection_butterfly|$resurrection_butterfly$|resurrection_butterfly_sp|$resurrection_butterfly_sp$|sakuya_watch|$sakuya_watch$|sanae|$sanae$|spark|$spark$|touhou_difficulty_player|$touhou_difficulty_player$|touhou_dynamic_damage_ag|$touhou_dynamic_damage_ag$|xianzhe|$xianzhe$|xieshi|$xieshi$|yuzhu|$yuzhu$|
    }
    solar_system = {
        every_ship_in_system = {
            limit = {
                NOT = { has_ship_flag = touhou_aoe_damage_template_$size$_$type$_lock }
                check_modifier_value = { modifier = ag_ship_aoe_damage_reduction value < 1.0 }
                check_modifier_value = { modifier = ship_aoe_damage_reduction_mult value < 1.0 }
                check_modifier_value = { modifier = touhou_ship_aoe_damage_reduction_mult value < 1.0 }
                exists = controller controller = { is_hostile = root.from.owner }
                distance = { same_solar_system = yes source = prevprev max_distance <= $aoe_range$ }
            }
            # i decide to not play explosion particle on every ship
            # this may cause tons tons of lag.
            set_timed_ship_flag = { flag = touhou_aoe_damage_template_$size$_$type$_lock days = $cooldown$ }
            set_variable = { which = touhou_aoe_damage_template_$size$_$type$_total_temp value = value:touhou_aoe_damage_template_final_value|touhou_base|prevprev.touhou_aoe_damage_template_$size$_$type$_total_self| }
            # calc shield damage here
            if = {
                limit = { has_shield_hp > 0 }
                set_variable = { which = touhou_aoe_damage_template_$size$_$type$_shield_temp value = touhou_aoe_damage_template_$size$_$type$_total_temp }
                multiply_variable = { which = touhou_aoe_damage_template_$size$_$type$_shield_temp value = prevprev.value:touhou_modifier_mult_multiplier|modifier|ship_shield_damage_mult| }
                if = {
                    limit = { has_shield_hp >= touhou_aoe_damage_template_$size$_$type$_shield_temp }
                    reduce_shield = touhou_aoe_damage_template_$size$_$type$_shield_temp
                }
                else = {
                    set_variable = { which = touhou_aoe_damage_template_$size$_$type$_shield_temp value = trigger:has_shield_hp }
                    divide_variable = { which = touhou_aoe_damage_template_$size$_$type$_shield_temp value = prevprev.value:touhou_modifier_mult_multiplier|modifier|ship_shield_damage_mult| }
                    reduce_shield = trigger:has_shield_hp
                }
                subtract_variable = { which = touhou_aoe_damage_template_$size$_$type$_total_temp value = touhou_aoe_damage_template_$size$_$type$_shield_temp }
                clear_variable = touhou_aoe_damage_template_$size$_$type$_shield_temp
            } else_if = { limit = { check_variable = { which = touhou_aoe_damage_template_$size$_$type$_total_temp value <= 0 } } clear_variable = touhou_aoe_damage_template_$size$_$type$_total_temp }
            # calc armor damage here
            if = {
                limit = { check_variable = { which = touhou_aoe_damage_template_$size$_$type$_total_temp value > 0 } has_armor_percentage > 0 }
                set_variable = { which = touhou_aoe_damage_template_ship_current_armor value = trigger:has_max_armor_hp }
                multiply_variable = { which = touhou_aoe_damage_template_ship_current_armor value = trigger:has_armor_percentage }
                set_variable = { which = touhou_aoe_damage_template_$size$_$type$_armor_temp value = touhou_aoe_damage_template_$size$_$type$_total_temp }
                multiply_variable = { which = touhou_aoe_damage_template_$size$_$type$_armor_temp value = prevprev.value:touhou_modifier_mult_multiplier|modifier|ship_armor_damage_mult| }
                if = {
                    limit = { check_variable = { which = touhou_aoe_damage_template_ship_current_armor value >= touhou_aoe_damage_template_$size$_$type$_armor_temp } }
                    multiply_variable = { which = touhou_aoe_damage_template_$size$_$type$_armor_temp value = -1 }
                    repair_armor_amount = touhou_aoe_damage_template_$size$_$type$_armor_temp
                }
                else = {
                    set_variable = { which = touhou_aoe_damage_template_$size$_$type$_armor_temp value = touhou_aoe_damage_template_ship_current_armor }
                    multiply_variable = { which = touhou_aoe_damage_template_$size$_$type$_armor_temp value = -1 }
                    repair_armor_amount = touhou_aoe_damage_template_$size$_$type$_armor_temp
                    set_variable = { which = touhou_aoe_damage_template_$size$_$type$_armor_temp value = touhou_aoe_damage_template_ship_current_armor }
                    divide_variable = { which = touhou_aoe_damage_template_$size$_$type$_armor_temp value = prevprev.value:touhou_modifier_mult_multiplier|modifier|ship_armor_damage_mult| }
                    multiply_variable = { which = touhou_aoe_damage_template_$size$_$type$_armor_temp value = -1 }
                }
                change_variable = { which = touhou_aoe_damage_template_$size$_$type$_total_temp value = touhou_aoe_damage_template_$size$_$type$_armor_temp }
                clear_variable = touhou_aoe_damage_template_ship_current_armor
                clear_variable = touhou_aoe_damage_template_$size$_$type$_armor_temp
                clear_variable = touhou_aoe_damage_template_$size$_$type$_armor_temp_alt
            }
            else_if = { limit = { check_variable = { which = touhou_aoe_damage_template_$size$_$type$_total_temp value <= 0 } } clear_variable = touhou_aoe_damage_template_$size$_$type$_total_temp }
            # calc hull damage here
            if = {
                limit = { check_variable = { which = touhou_aoe_damage_template_$size$_$type$_total_temp value > 0 } }
                multiply_variable = { which = touhou_aoe_damage_template_$size$_$type$_total_temp value = prevprev.value:touhou_modifier_mult_multiplier|modifier|ship_hull_damage_mult| }
                if = { limit = { has_modifier = $hull_damage_modifier$ } remove_modifier = $hull_damage_modifier$ }
                add_modifier = { modifier = $hull_damage_modifier$ days = 2 multiplier = touhou_aoe_damage_template_$size$_$type$_total_temp }
            }
            clear_variable = touhou_aoe_damage_template_$size$_$type$_total_temp
        }
    }
    inline_script = {
        script = effect/touhou_spawn_explosion_particle_$spawn_hit_effect$
        color = $effect_color$ size = $effect_size$
    }
    clear_variable = touhou_aoe_damage_template_$size$_$type$_total_self
}
