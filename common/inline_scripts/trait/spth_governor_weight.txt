selectable_weight = {
    weight = $base$
    modifier = {
        exists =  planet
        factor = @job_weight_mult
    }
    modifier = {
        NOT = { exists =  planet }
        factor = @job_weight_opp_mult
    }
    modifier = {
        has_governor_subclass = yes
        factor = @subclass_weight_mult
    }
    modifier = {
        AND = {
            has_paragon_dlc = yes
            has_total_skill > 3
            has_governor_subclass = no
        }
        factor = @subclass_weight_opp_mult
    }
    modifier = {
        has_leader_flag = block_governor_traits
        has_governor_subclass = no
        factor = 0
    }

}
