		}

		# Ethic does not match (non-gestalt)
		add = value:touhou_ai_ideology_affinity_from_non_matching_ethic

			has_same_ethic_as_ideology = from

		}
	}
	modifier = {
		# Ethic does not match (gestalt)
		add = -1
		is_gestalt = yes
		ideology? = {
			NOR = {
				has_same_ethic_as_ideology = root.from
				has_opposite_ethic_as_ideology = root.from
			}
		}

	}
	modifier = {
		# Gestalt base
		add = -2.5
		is_gestalt = yes
	}
}

	every_federation = {
		limit = {
			ideology? = { is_same_value = root }

		}
		set_federation_law = succession_type_none
	}

	every_playable_country = {
		clear_variable = pops_under_reach_of_@root
	}
