touhou_mirror_empire_create_country = {
    optimize_memory
    if = {
        limit = { NOT = { exists = event_target:touhou_mirror_empire } }
        create_species = {
            name = "NAME_touhou_mirror_empire_species"
            class = THSC
            portrait = touhou2
            traits = random
            immortal = yes
            effect = {
                save_global_event_target_as = touhou_mirror_empire_species
            }
        }
        create_country = {
            name = touhou_mirror_empire
            type = touhou_mirror_empire
            species = event_target:touhou_mirror_empire_species
            flag = event_target:touhou_mirror_empire_root
            name_list = event_target:touhou_mirror_empire_root
            effect = {
                set_graphical_culture = event_target:touhou_mirror_empire_root
                set_country_flag = touhou_mod_country
                set_country_flag = touhou_crisis_country
                set_country_flag = touhou_mirror_empire
                set_country_flag = touhou_mirror_empire_by_@event_target:touhou_mirror_empire_root
                set_country_flag = protected_from_queen_storm
                add_modifier = { modifier = touhou_immune_from_terrify }
                save_global_event_target_as = touhou_mirror_empire
                touhou_mirror_empire_switch_tech_level = yes
                if = {
                    limit = { touhou_mirror_empire_check_tech_level = { relation = "=" value = 3 } }
                    create_ship_design = { design = NAME_touhou_mirror_empire_starbase_3 }
                    add_ship_design = last_created_design
                } else_if = {
                    limit = { touhou_mirror_empire_check_tech_level = { relation = "=" value = 1 } }
                    create_ship_design = { design = NAME_touhou_mirror_empire_starbase_1 }
                    add_ship_design = last_created_design
                } else_if = {
                    limit = { touhou_mirror_empire_check_tech_level = { relation = "=" value = 2 } }
                    create_ship_design = { design = NAME_touhou_mirror_empire_starbase_2 }
                    add_ship_design = last_created_design
                } else = {
                    create_ship_design = { design = NAME_touhou_mirror_empire_starbase }
                    add_ship_design = last_created_design
                }
                # activate_fog_machine = spth_kamikakushi_cloud
                # fog_machine_auto_tracking = yes
                [[debug_spawn_flag] set_country_flag = spth_debug_spawn_flag ]
                every_playable_country = {
                    limit = { NOT = { is_same_value = prev } }
                    establish_communications_no_message = prev
                    add_intel = { who = prev amount = 70 }
                }
                every_country = {
                    limit = {
                        OR = {
                            is_country_type = default
                            is_country_type = fallen_empire
                            is_country_type = awakened_fallen_empire
                        }
                        NOT = { has_communications = prev }
                    }
                    establish_communications_no_message = prev
                    add_intel = { who = prev amount = 70 }
                }
                country_event = { id = touhou_mirror_empire.20 days = 360 random = 180 }
            }
        }
    }
}

touhou_mirror_empire_debug_set_root_country = {
    optimize_memory
    if = {
        limit = { NOT = { exists = event_target:touhou_mirror_empire_root } }
        if = {
            limit = { NOT = { has_global_flag = touhou_mirror_empire_no_root_country } }
            if = {
                limit = { any_playable_country = { has_authority = auth_gensokyo is_ai = no } }
                random_playable_country = {
                    limit = { has_authority = auth_gensokyo is_ai = no }
                    save_global_event_target_as = touhou_mirror_empire_root
                }
            } else_if = {
                limit = { any_playable_country = { has_authority = auth_gensokyo is_ai = yes } }
                random_playable_country = {
                    limit = { has_authority = auth_gensokyo is_ai = yes }
                    save_global_event_target_as = touhou_mirror_empire_root
                }
            } else_if = {
                limit = { any_playable_country = { is_ai = no } }
                random_playable_country = {
                    limit = { is_ai = no }
                    save_global_event_target_as = touhou_mirror_empire_root
                }
            } else = {
                set_global_flag = touhou_mirror_empire_no_root_country
            }
        }
    }
}

touhou_mirror_empire_switch_tech_level = {
    optimize_memory
    if = {
        limit = { NOT = { has_global_flag = touhou_mirror_empire_switch_tech_level_cooldown } }
        set_timed_global_flag = { flag = touhou_mirror_empire_switch_tech_level_cooldown days = 5 }
        event_target:touhou_mirror_empire = {
            if = {
                limit = { NOT = { is_variable_set = touhou_mirror_empire_tech_level } }
                touhou_mirror_empire_set_tech_level = { level = 0 }
            }
        }
        if = {
            limit = { exists = event_target:touhou_mirror_empire_root }
            event_target:touhou_mirror_empire_root = {
                if = {
                    limit = { spth_has_ag_mod = yes }
                    if = {
                        limit = { touhou_mirror_empire_check_tech_level = { relation = "=" value = 3 } }
                        break = yes
                    } else_if = {
                        limit = {
                            OR = {
                                count_owned_ship = { limit = { ag_is_ancient_titan = yes } count >= @touhou_mirror_empire_tech_level_ag_titan_count }
                                count_owned_ship = { limit = { ag_is_ancient_ship = yes } count >= @touhou_mirror_empire_tech_level_ag_ship_count }
                            }
                        }
                        touhou_mirror_empire_set_tech_level = { level = 3 }
                        break = yes
                    }
                    if = {
                        limit = { touhou_mirror_empire_check_tech_level = { relation = "=" value = 1 } }
                        break = yes
                    } else_if = {
                        limit = { has_technology = tech_lunar_capital_parts }
                        touhou_mirror_empire_set_tech_level = { level = 1 }
                        break = yes
                    }
                    if = {
                        limit = { touhou_mirror_empire_check_tech_level = { relation = "=" value = 2 } }
                        break = yes
                    } else_if = {
                        limit = {
                            calc_true_if = {
                                amount >= 3
                                has_technology = tech_th_armor_scarlet_psionic
                                has_technology = tech_th_shield_barrier_psionic
                                OR = {
                                    has_technology = tech_th_shield_barrier_psionic_story
                                    has_technology = tech_th_armor_barrier_psionic_story
                                }
                                has_technology = tech_touhou_psionic_lighting
                                has_technology = tech_touhou_psionic_lighting_pd
                                has_technology = tech_touhou_psionic_lighting_xl
                                has_technology = tech_touhou_psionic_lighting_t
                                has_technology = tech_trait_touhou_psionic
                                has_technology = tech_th_ship_part_mokou
                            }
                        }
                        touhou_mirror_empire_set_tech_level = { level = 2 }
                        break = yes
                    }
                    if = {
                        limit = { touhou_mirror_empire_check_tech_level = { relation = "!=" value = 0 } }
                        touhou_mirror_empire_set_tech_level = { level = 0 }
                        break = yes
                    }
                } else = {
                    if = {
                        limit = { touhou_mirror_empire_check_tech_level = { relation = "=" value = 1 } }
                        break = yes
                    } else_if = {
                        limit = { has_technology = tech_lunar_capital_parts }
                        touhou_mirror_empire_set_tech_level = { level = 1 }
                        break = yes
                    }
                    if = {
                        limit = { touhou_mirror_empire_check_tech_level = { relation = "=" value = 2 } }
                        break = yes
                    } else_if = {
                        limit = {
                            calc_true_if = {
                                amount >= 3
                                has_technology = tech_th_armor_scarlet_psionic
                                has_technology = tech_th_shield_barrier_psionic
                                OR = {
                                    has_technology = tech_th_shield_barrier_psionic_story
                                    has_technology = tech_th_armor_barrier_psionic_story
                                }
                                has_technology = tech_touhou_psionic_lighting
                                has_technology = tech_touhou_psionic_lighting_pd
                                has_technology = tech_touhou_psionic_lighting_xl
                                has_technology = tech_touhou_psionic_lighting_t
                                has_technology = tech_trait_touhou_psionic
                                has_technology = tech_th_ship_part_mokou
                            }
                        }
                        touhou_mirror_empire_set_tech_level = { level = 2 }
                        break = yes
                    }
                    if = {
                        limit = { touhou_mirror_empire_check_tech_level = { relation = "!=" value = 0 } }
                        touhou_mirror_empire_set_tech_level = { level = 0 }
                        break = yes
                    }
                }
            }
        }
    }
}
