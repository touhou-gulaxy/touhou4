spth_spawn_lunar_capital_union_fleet = {
    optimize_memory
    if = {
        limit = { NOT = { event_target:global_event_country = { is_variable_set = spth_lunar_capital_crisis_difficulty } } }
        event_target:global_event_country = { set_variable = { which = spth_lunar_capital_crisis_difficulty value = 1 } }
    }
    set_update_modifiers_batch = begin
    if = {
        limit = { NOT = { exists = event_target:spth_lunar_capital_union_fleet_country } }
        create_country = {
            name = spth_lunar_union_fleet_country
            type = spth_lunar_capital_union_fleet
            flag = {
                icon = {
                    category = "touhousp"
                    file = "lunar_capital_alt.dds"
                }
                colors = {
                    "blue"
                    "blue"
                    "blue"
                    "null"
                }
                background = {
                    category = "backgrounds"
                    file = "00_solid.dds"
                }
            }
            name_list = touhouchs
            effect = {
                set_graphical_culture = strg_01
                set_country_flag = touhou_mod_country
                set_country_flag = spth_lunar_capital_union_fleet_country
                add_modifier = { modifier = touhou_immune_from_terrify }
                [[debug_spawn_flag] set_country_flag = spth_debug_spawn_flag ]
                save_global_event_target_as = spth_lunar_capital_union_fleet_country
                every_playable_country = {
                    limit = { NOT = { is_same_value = prev } }
                    establish_communications_no_message = prev
                }
            }
        }
    }
    event_target:spth_story_lunar_capital_union_system = {
        spth_create_fleet = {
            NAME = spth_lunar_union_fleet
            can_disband = no is_boss = yes
            OWNER = event_target:spth_lunar_capital_union_fleet_country
            location = prev uses_naval_capacity = no
            flag = spth_lunar_capital_union_fleet
            NAME_lunar_capital_battleship_crisis = 2
            NAME_lunar_capital_escort_crisis = 6
            NAME_lunar_capital_frigate_crisis = 8
            NAME_lunar_capital_frigate_s_crisis = 8
            evt_target = spth_lunar_capital_union_fleet
            lunar_capital_bonus = yes
        }
    }
    set_update_modifiers_batch = end
}

spth_lunar_psi_setup_commander = {
    optimize_memory
    if = {
        limit = { has_global_flag = has_spth_lunar_capital_psi_species }
        event_target:spth_lunar_capital_psi_country = {
            create_leader = {
                name = NAME_lunar_capital_shroud
                class = commander
                species = event_target:spth_lunar_capital_psi_species
                skip_background_generation = yes
                custom_catch_phrase = lunar_capital_shroud_catch
                custom_description = lunar_capital_shroud_bg
                skill = 4
                traits = {
                    trait = leader_trait_lunar_capital
                    trait = leader_trait_lunar_capital_psionic
                }
                effect = {
                    set_leader_flag = spth_lunar_capital_leader
                    set_leader_flag = touhou_shroud_leader
                    set_leader_flag = has_gained_level_trait
                    set_leader_flag = legendary_leader
                    add_trait = { trait = subclass_commander_admiral }
                }
            }
            last_created_leader = {
                add_trait = { trait = leader_trait_artillery_specialization_2 show_message = no }
                add_trait = { trait = leader_trait_artillerist_2 show_message = no }
                # 额外技能等级，仅HARD及以上
                [[!force_level]
                    if = {
                        limit = { touhou_check_diffiulty_gteq = { value = 3 } }
                        add_skill = 3
                        if = {
                            limit = { touhou_check_diffiulty_gteq = { value = 4 } }
                            add_skill = 170
                        }
                    }
                ]
                [[force_level]
                    add_skill = $force_level$
                ]
                if = {
                    limit = { has_total_skill >= 8 }
                    reroll_random = yes
                    random_list = {
                        10 = { add_trait = { trait = leader_trait_survivalist show_message = no } }
                        10 = { add_trait = { trait = leader_trait_guerilla_tactics show_message = no } }
                        10 = { add_trait = { trait = leader_trait_master_gunner show_message = no } }
                        10 = { add_trait = { trait = leader_trait_destiny_engineer show_message = no } }
                    }
                }
                save_event_target_as = last_created_touhou_crisis_leader
            }
            event_target:$fleet$ = {
                assign_leader = event_target:last_created_touhou_crisis_leader
            }
        }
    }
}

spth_lunar_psi_pick_mirror_design = {
    optimize_memory
    if = {
        limit = { any_playable_country = { has_authority = auth_gensokyo is_ai = no } }
        random_playable_country = {
            limit = { has_authority = auth_gensokyo is_ai = no }
            save_global_event_target_as = touhou_pick_mirror_country
        }
    } else_if = {
        limit = { any_playable_country = { has_authority = auth_gensokyo } }
        random_playable_country = {
            limit = { has_authority = auth_gensokyo }
            save_global_event_target_as = touhou_pick_mirror_country
        }
    } else = {
        random_playable_country = {
            save_global_event_target_as = touhou_pick_mirror_country
        }
    }
    if = {
        limit = { NOT = { exists = event_target:touhou_pick_mirror_country } }
        set_timed_global_flag = { flag = no_lunar_capital_pick_mirror_country days = 180 }
    } else = {
        set_global_flag = spth_lunar_psi_picked_mirror_design
        event_target:touhou_pick_mirror_country = {
            if = {
                limit = { count_owned_fleet = { limit = { is_touhou_cannot_mirror_fleet = no } count > 0 } }
                random_owned_fleet = {
                    limit = { is_touhou_cannot_mirror_fleet = no }
                    random_owned_ship = {
                        limit = { is_touhou_cannot_mirror_ship = no }
                        weights = {
                            base = 10
                            modifier = { factor = 0.1 design = { has_design_flag = was_touhou_mirror_design } }
                        }
                        design = {
                            save_global_event_target_as = touhou_mirror_design
                            set_design_flag = touhou_mirror_design
                            set_design_flag = was_touhou_mirror_design
                        }
                        event_target:spth_lunar_capital_psi_country = {
                            set_variable = {
                                which = touhou_mirror_design_ship_size
                                value = prev.trigger:count_used_naval_cap
                            }
                        }
                    }
                }
            } else = {
                random_owned_design = {
                    limit = { is_ship_class = shipclass_military is_designable = yes }
                    weights = {
                        base = 10
                        modifier = { factor = 0.1 has_design_flag = was_touhou_mirror_design }
                    }
                    save_global_event_target_as = touhou_mirror_design
                    set_design_flag = touhou_mirror_design
                    set_design_flag = was_touhou_mirror_design
                    event_target:spth_lunar_capital_psi_country = {
                        set_variable = {
                            which = touhou_mirror_design_ship_size
                            value = prev.value:spth_get_ship_size
                        }
                    }
                }
            }
        }
        event_target:spth_lunar_capital_psi_country = {
            set_variable = {
                which = lunar_capital_psi_mirror_ship_count
                value = value:lunar_capital_psi_mirror_ship_count
            }
            set_variable = {
                which = lunar_capital_psi_mirror_damage_reduction
                value = value:lunar_capital_psi_mirror_damage_reduction
            }
            add_modifier = {
                modifier = touhou_lunar_psi_extra_41
                mult = lunar_capital_psi_mirror_damage_reduction
            }
        }
    }
}
