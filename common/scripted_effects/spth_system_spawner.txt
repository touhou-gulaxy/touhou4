create_lunar_capital_fleet = {
    optimize_memory
    create_fleet = {
        name = $NAME$
        settings = {
            can_change_composition = yes
            uses_naval_capacity = no
            spawn_debris = no
            can_disband = $can_disband$
            is_boss = $is_boss$
        }
        effect = {
            set_owner = $OWNER$
            [[bombardment] set_fleet_bombardment_stance = $bombardment$ ]
            [[flag] set_fleet_flag = $flag$ ]
            [[star_sector_flagship]
                while = {
                    count = $star_sector_flagship$
                    create_ship = {
                        name = random design = NAME_star_sector_flagship
                        effect = {
                            set_ship_flag = has_ag_flagship
                            set_ship_flag = has_sg_flagship
                            set_ship_flag = has_ag_sub_flagship
                        }
                    }
                }
            ]
            set_location = $location$
        }
    }
}

spth_make_kamikakushi_fallen_empire_awake = {
    optimize_memory
    set_country_flag = spth_awaken_empire_kamikakushi
    set_country_type = spth_awaken_empire_kamikakushi
    spth_refresh_kamikakushi_awaken_empire_fleets = yes
    unlock_council_slots = 2
    while = {
        count = 2
        create_leader = {
            class = scientist
            species = owner_main_species
            name = random
            skill = 10
            leader_age_min = 25
            leader_age_max = 45
        }
    }
    # Establish communications
    [[threat]
        set_country_flag = spth_awaken_by_threat
        $threat$ = { save_event_target_as = enemy_empire }
    ]
    every_country = {
        limit = {
            OR = {
                is_country_type = default
                is_country_type = fallen_empire
                is_country_type = awakened_fallen_empire
            }
            NOT = { has_communications = root }
        }
        establish_communications_no_message = root
        root = {
            save_event_target_as = contact_empire
        }
        country_event = { id = spth_precursor.204 }
    }
}
