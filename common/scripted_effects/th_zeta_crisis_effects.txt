th_zeta_crisis_space_monster_swap = {
    optimize_memory
    if = {
        limit = { any_owned_ship = { is_ship_size = spth_kedama_small } }
        create_fleet = {
            settings = { spawn_debris = no }
            effect = {
                set_name = prev
                set_owner = event_target:ag_zeta_crisis_country
                set_fleet_flag = ag_zeta_space_monster_fleet
                prev = {
                    every_owned_ship = {
                        limit = { is_ship_size = spth_kedama_small }
                        switch = {
                            trigger = is_ship_size
                            spth_kedama_small = { zeta_crisis_touhou_kedama_swap = yes }
                        }
                        delete_ship = this
                    }
                }
                set_location = { target = prev distance = 0 angle = 0 }
            }
        }
    }
}
