touhou_generate_ideology = {
	optimize_memory
	set_country_flag = ideology_initiated # Just to make sure
	if = {
		limit = { is_ai = yes }
		set_country_flag = cannot_codify_ideology
	}
	create_ideology = {
		[[ethic]
		ethic = $ethic$
		]
		[[faction]
		faction = $faction$
		]
		[[name]
		name = $name$
		]
		[[type]
		type = $type$
		]
		[[tenets]
		tenets = { $tenets$ }
		]
		[[tenets_from]
		tenets = $tenets_from$
		]
		effect = {
			save_event_target_as = generated_ideology
			create_message = {
				type = MESSAGE_IDEOLOGY_EMERGED
				localization = MESSAGE_IDEOLOGY_EMERGED_DESC
				days = 90
				onclick = ideologies
			}
		}
	}
}
