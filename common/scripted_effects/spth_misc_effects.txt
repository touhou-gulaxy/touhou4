touhou_flower_incident_system = {
    optimize_memory
    if = {
        limit = { NOT = { has_star_flag = affect_by_touhou_flower_incident } }
        set_star_flag = affect_by_touhou_flower_incident
        add_event_chain_counter = {
            event_chain = touhou_flower_incident
            counter = touhou_flower_incident_our_affect_system
            amount = 1
        }
        add_event_chain_counter = {
            event_chain = touhou_flower_incident
            counter = touhou_flower_incident_our_ghost_affect_planet
            amount = trigger:count_system_colony
        }
    }
    remove_star_flag = touhou_flower_incident_stage_1
    remove_star_flag = touhou_flower_incident_stage_2
    remove_star_flag = touhou_flower_incident_stage_3
    set_star_flag = touhou_flower_incident_stage_$level$
    every_system_colony = {
        limit = { NOT = { has_planet_flag = touhou_flower_incident_stage_$level$ } }
        remove_planet_flag = touhou_flower_incident_stage_1
        remove_planet_flag = touhou_flower_incident_stage_2
        remove_planet_flag = touhou_flower_incident_stage_3
        set_planet_flag = touhou_flower_incident_stage_$level$
        touhou_flower_incident_colony_modifier = { level = $level$ }
    }
}
touhou_flower_incident_spread = {
    optimize_memory
    if = {
        limit = { NOT = { has_country_flag = touhou_flower_incident_spreaded } }
        set_country_flag = touhou_flower_incident_spreaded
    }
    reroll_random = yes
    random_list = {
        75 = {
            every_system_within_border = {
                limit = {
                    NOT = { has_star_flag = touhou_flower_incident_stage_3 }
                    any_neighbor_system = { has_star_flag = affect_by_touhou_flower_incident }
                }
                if = {
                    limit = { has_star_flag = touhou_flower_incident_stage_2 }
                    touhou_flower_incident_system = { level = 3 }
                } else_if = {
                    limit = { has_star_flag = touhou_flower_incident_stage_1 }
                    touhou_flower_incident_system = { level = 2 }
                }
            }
        }
        25 = {}
    }
    reroll_random = yes
    random_system_within_border = {
        limit = {
            NOT = { has_star_flag = affect_by_touhou_flower_incident }
            any_neighbor_system = { has_star_flag = affect_by_touhou_flower_incident }
        }
        touhou_flower_incident_system = { level = 1 }
    }
}

touhou_ac_research_technology = {
    optimize_memory
    if = {
        limit = {
            check_variable_arithmetic = {
                which = value:touhou_ac_research_cost
                subtract = value:touhou_ac_research_base_normalized
                value > th_weapon_up_research_progress
            }
        }
        change_variable = { which = th_weapon_up_research_progress value = value:touhou_ac_research_base_normalized }
    } else = {
        subtract_variable = { which = th_weapon_up_research_progress value = value:touhou_ac_research_cost }
        change_variable = { which = th_weapon_up_research_progress value = value:touhou_ac_research_base_normalized }
        if = {
            limit = { check_variable = { which = th_weapon_up_research_progress value > 0 } }
            touhou_ac_research_technology_item = yes
            if = {
                limit = {
                    check_variable_arithmetic = {
                        which = value:touhou_ac_research_cost
                        subtract = value:touhou_ac_research_base_normalized
                        value > th_weapon_up_research_progress
                    }
                }
                set_variable = { which = th_weapon_up_research_progress value = 0 }
                touhou_ac_research_technology_item = yes
            } else = { set_variable = { which = th_weapon_up_research_progress value = 0 } }
        } else = { set_variable = { which = th_weapon_up_research_progress value = 0 } }
    }
}

touhou_ac_refresh_research_point_using = {
    optimize_memory
    if = {
        limit = { NOT = { is_variable_set = touhou_ac_$type$_research_using } }
        set_variable = { which = touhou_ac_$type$_research_using value = 0 }
        export_resource_income_to_variable = { resource = $type$_research variable = touhou_ac_$type$_research_using }
        multiply_variable = { which = touhou_ac_$type$_research_using value = $factor$ }
        add_modifier = { modifier = touhou_ac_$type$_research_using days = -1 mult = touhou_ac_$type$_research_using }
    } else = {
        set_variable = { which = touhou_ac_$type$_research_using_alt value = 0 }
        export_resource_income_to_variable = { resource = $type$_research variable = touhou_ac_$type$_research_using_alt }
        change_variable = { which = touhou_ac_$type$_research_using_alt value = touhou_ac_$type$_research_using }
        set_variable = { which = touhou_ac_$type$_research_using value = touhou_ac_$type$_research_using_alt }
        clear_variable = touhou_ac_$type$_research_using_alt
        multiply_variable = { which = touhou_ac_$type$_research_using value = $factor$ }
        while = { limit = { has_modifier = touhou_ac_$type$_research_using } remove_modifier = touhou_ac_$type$_research_using }
        add_modifier = { modifier = touhou_ac_$type$_research_using days = -1 mult = touhou_ac_$type$_research_using }
    }
}

touhou_destory_crisis_shpere = {
    optimize_memory
    if = {
        limit = { has_global_flag = touhou_lunar_psi_extra_33 }
        every_megastructure = {
            limit = {
                OR = {
                    is_megastructure_type = crisis_sphere_0
                    is_megastructure_type = crisis_sphere_1
                    is_megastructure_type = crisis_sphere_2
                    is_megastructure_type = crisis_sphere_3
                    is_megastructure_type = crisis_sphere_4
                    is_megastructure_type = crisis_sphere_ruined
                }
            }
            solar_system = {
                every_system_ambient_object = {
                    limit = {
                        OR = {
                            is_ambient_object_type = crisis_sphere_1
                            is_ambient_object_type = crisis_sphere_2
                            is_ambient_object_type = crisis_sphere_3
                        }
                    }
                    destroy_ambient_object = this
                }
            }
            if = {
                limit = { exists = owner }
                solar_system = {
                    spawn_megastructure = {
                        type = touhou_dummy_crisis_sphere_ruined
                        coords_from = prev
                        owner = prev.owner
                    }
                    save_event_target_as = spth_target_system
                }
                if = {
                    limit = { NAND = { is_megastructure_type = crisis_sphere_ruined owner = { NOT = { has_crisis_level = crisis_level_5 } } } }
                    owner = { country_event = { id = lunar_capital_psi.34 days = 2 random = 2 } }
                }
            } else = {
                solar_system = {
                    spawn_megastructure = {
                        type = touhou_dummy_crisis_sphere_ruined
                        coords_from = prev
                    }
                }
            }
            th_spawn_explosion_particle = { size = extra color = purple }
            remove_megastructure = this
        }
    }
}
