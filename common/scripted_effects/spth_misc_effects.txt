touhou_flower_incident_system = {
    optimize_memory
    if = {
        limit = { NOT = { has_star_flag = affect_by_touhou_flower_incident } }
        set_star_flag = affect_by_touhou_flower_incident
        add_event_chain_counter = {
            event_chain = touhou_flower_incident
            counter = touhou_flower_incident_our_affect_system
            amount = 1
        }
        add_event_chain_counter = {
            event_chain = touhou_flower_incident
            counter = touhou_flower_incident_our_ghost_affect_planet
            amount = trigger:count_system_colony
        }
    }
    remove_star_flag = touhou_flower_incident_stage_1
    remove_star_flag = touhou_flower_incident_stage_2
    remove_star_flag = touhou_flower_incident_stage_3
    set_star_flag = touhou_flower_incident_stage_$level$
    every_system_colony = {
        limit = { NOT = { has_planet_flag = touhou_flower_incident_stage_$level$ } }
        remove_planet_flag = touhou_flower_incident_stage_1
        remove_planet_flag = touhou_flower_incident_stage_2
        remove_planet_flag = touhou_flower_incident_stage_3
        set_planet_flag = touhou_flower_incident_stage_$level$
        touhou_flower_incident_colony_modifier = { level = $level$ }
    }
}

touhou_ac_research_technology = {
    optimize_memory
    if = {
        limit = {
            check_variable_arithmetic = {
                which = value:touhou_ac_research_cost
                subtract = value:touhou_ac_research_base_normalized
                value > th_weapon_up_research_progress
            }
        }
        change_variable = { which = th_weapon_up_research_progress value = value:touhou_ac_research_base_normalized }
    } else = {
        subtract_variable = { which = th_weapon_up_research_progress value = value:touhou_ac_research_cost }
        change_variable = { which = th_weapon_up_research_progress value = value:touhou_ac_research_base_normalized }
        if = {
            limit = { check_variable = { which = th_weapon_up_research_progress value > 0 } }
            touhou_ac_research_technology_item = yes
            if = {
                limit = {
                    check_variable_arithmetic = {
                        which = value:touhou_ac_research_cost
                        subtract = value:touhou_ac_research_base_normalized
                        value > th_weapon_up_research_progress
                    }
                }
                set_variable = { which = th_weapon_up_research_progress value = 0 }
                touhou_ac_research_technology_item = yes
            } else = { set_variable = { which = th_weapon_up_research_progress value = 0 } }
        } else = { set_variable = { which = th_weapon_up_research_progress value = 0 } }
    }
}

touhou_ac_refresh_research_point_using = {
    optimize_memory
    if = {
        limit = { NOT = { is_variable_set = touhou_ac_$type$_research_using } }
        set_variable = { which = touhou_ac_$type$_research_using value = 0 }
        export_resource_income_to_variable = { resource = $type$_research variable = touhou_ac_$type$_research_using }
        multiply_variable = { which = touhou_ac_$type$_research_using value = $factor$ }
        add_modifier = { modifier = touhou_ac_$type$_research_using days = -1 mult = touhou_ac_$type$_research_using }
    } else = {
        set_variable = { which = touhou_ac_$type$_research_using_alt value = 0 }
        export_resource_income_to_variable = { resource = $type$_research variable = touhou_ac_$type$_research_using_alt }
        change_variable = { which = touhou_ac_$type$_research_using_alt value = touhou_ac_$type$_research_using }
        set_variable = { which = touhou_ac_$type$_research_using value = touhou_ac_$type$_research_using_alt }
        clear_variable = touhou_ac_$type$_research_using_alt
        multiply_variable = { which = touhou_ac_$type$_research_using value = $factor$ }
        while = { limit = { has_modifier = touhou_ac_$type$_research_using } remove_modifier = touhou_ac_$type$_research_using }
        add_modifier = { modifier = touhou_ac_$type$_research_using days = -1 mult = touhou_ac_$type$_research_using }
    }
}

touhou_destory_crisis_shpere = {
    optimize_memory
    if = {
        limit = { has_global_flag = touhou_lunar_psi_extra_33 }
        every_megastructure = {
            limit = {
                OR = {
                    is_megastructure_type = crisis_sphere_0
                    is_megastructure_type = crisis_sphere_1
                    is_megastructure_type = crisis_sphere_2
                    is_megastructure_type = crisis_sphere_3
                    is_megastructure_type = crisis_sphere_4
                    is_megastructure_type = crisis_sphere_ruined
                }
            }
            solar_system = {
                every_system_ambient_object = {
                    limit = {
                        OR = {
                            is_ambient_object_type = crisis_sphere_1
                            is_ambient_object_type = crisis_sphere_2
                            is_ambient_object_type = crisis_sphere_3
                        }
                    }
                    destroy_ambient_object = this
                }
            }
            if = {
                limit = { exists = owner }
                solar_system = {
                    spawn_megastructure = {
                        type = touhou_dummy_crisis_sphere_ruined
                        coords_from = prev
                        owner = prev.owner
                    }
                    save_event_target_as = spth_target_system
                }
                if = {
                    limit = { NAND = { is_megastructure_type = crisis_sphere_ruined owner = { NOT = { has_crisis_level = crisis_level_5 } } } }
                    owner = { country_event = { id = lunar_capital_psi.34 days = 2 random = 2 } }
                }
            } else = {
                solar_system = {
                    spawn_megastructure = {
                        type = touhou_dummy_crisis_sphere_ruined
                        coords_from = prev
                    }
                }
            }
            th_spawn_explosion_particle = { size = extra color = purple }
            remove_megastructure = this
        }
    }
}

touhou_leader_special_army_effect_loop = {
    optimize_memory
    if = {
        limit = { exists = event_target:touhou_leader_kill_army_chance_$who$ }
        reroll_random = yes
        random_list = {
            1 = {
                modifier = { add = event_target:touhou_leader_kill_army_chance_$who$.modifier:touhou_leader_kill_army_chance }
                random_ground_combat_$opposite_who$ = { limit = { is_in_frontline = yes } remove_army = yes }
                log = "kill army check passed"
            }
            99 = { log = "kill army check failed" }
        }
    }
    if = {
        limit = { exists = event_target:touhou_leader_kill_army_perc_$who$ }
        reroll_random = yes
        random_list = {
            5 = {
                modifier = {
                    mult = event_target:touhou_leader_kill_army_perc_$who$.trigger:has_total_skill
                    event_target:touhou_leader_kill_army_perc_$who$ = { has_total_skill <= 19 }
                }
                modifier = { mult = 19 event_target:touhou_leader_kill_army_perc_$who$ = { has_total_skill > 19 } }
                while = {
                    count = value:touhou_leader_kill_army_perc_amount|who|$who$|opposite_who|$opposite_who$|
                    random_ground_combat_$opposite_who$ = { remove_army = yes }
                }
                log = "kill prec check passed"
            }
            95 = {
                modifier = { subtract = value:touhou_leader_kill_army_perc_minus|who|$who$| }
                log = "kill prec check failed"
            }
        }
    }
    if = {
        limit = { exists = event_target:touhou_leader_damage_army_$who$ }
        random_ground_combat_$opposite_who$ = {
            limit = { is_in_frontline = yes }
            damage_army = event_target:touhou_leader_damage_army_$who$.modifier:touhou_leader_damage_army_effect
        }
    }
}

touhou_weapon_fire_aoe_damage = {
    optimize_memory
    if = {
        limit = {
            check_modifier_value = { modifier = has_ship_part_aura_touhou_minefield_target value != 0 }
            exists = fleet fleet = { NOT = { has_fleet_flag = touhou_aoe_damage_$size$_$type$ } }
        }
        fleet = { set_timed_fleet_flag = { flag = touhou_aoe_damage_$size$_$type$ days = @touhou_aoe_damage_$size$_cooldown } }
        if = {
            limit = { exists = from }
            set_variable = {
                which = touhou_aoe_damage_$size$_$type$_hull
                value = from.value:touhou_aoe_hit_damage|size|$size$|tag1|$tag$|tag2|$type$|size_tag|$size_tag$|layer|hull|
            }
            set_variable = {
                which = touhou_aoe_damage_$size$_$type$_armor
                value = from.value:touhou_aoe_hit_damage|size|$size$|tag1|$tag$|tag2|$type$|size_tag|$size_tag$|layer|armor|
            }
            set_variable = { which = touhou_aoe_damage_$size$_$type$_armor_neg value = touhou_aoe_damage_$size$_$type$_armor }
            multiply_variable = { which = touhou_aoe_damage_$size$_$type$_armor_neg value = -1 }
            set_variable = {
                which = touhou_aoe_damage_$size$_$type$_shield
                value = from.value:touhou_aoe_hit_damage|size|$size$|tag1|$tag$|tag2|$type$|size_tag|$size_tag$|layer|shield|
            }
            fleet = {
                every_owned_ship = {
                    set_timed_ship_flag = { flag = touhou_aoe_damage_$size$_$type$ days = @touhou_aoe_damage_$size$_cooldown }
                    if = {
                        limit = { prevprev = { check_variable = { which = touhou_aoe_damage_$size$_$type$_armor value <= prev.trigger:has_shield_hp } } }
                        reduce_shield = prevprev.touhou_aoe_damage_$size$_$type$_shield
                    } else_if = {
                        limit = { prevprev = { check_variable = { which = touhou_aoe_damage_$size$_$type$_armor value <= prev.value:spth_armor_value } } }
                        repair_armor_amount = prevprev.touhou_aoe_damage_$size$_$type$_armor_neg
                    } else = {
                        add_modifier = { modifier = weapon_$type$_$size$_hull_damage days = 2 mult = prevprev.touhou_aoe_damage_$size$_$type$_hull }
                    }
                }
            }
            clear_variable = touhou_aoe_damage_$size$_$type$_armor_neg
        } else = {
            fleet = {
                every_owned_ship = {
                    set_timed_ship_flag = { flag = touhou_aoe_damage_$size$_$type$ days = @touhou_aoe_damage_$size$_cooldown }
                    if = {
                        limit = { has_shield_hp >= @touhou_aoe_damage_$size$ }
                        reduce_shield = @touhou_aoe_damage_$size$
                    } else_if = {
                        limit = { check_variable_arithmetic = { which = value:spth_armor_value value >= @touhou_aoe_damage_$size$ } }
                        repair_armor_amount = @touhou_aoe_damage_$size$_neg
                    } else = {
                        add_modifier = { modifier = weapon_$type$_$size$_hull_damage days = 2 mult = @touhou_aoe_damage_$size$ }
                    }
                }
            }
        }
    }
}

touhou_create_shell_world_layer = {
    optimize_memory
    set_planet_flag = has_megastructure
    save_event_target_as = ag_shell_world_prev_layer
    set_planet_entity = { entity = "ag_void_entity" }
    if = {
        limit = { ag_is_shell_world = no }
        set_variable = { which = ag_shell_world_layer value = 0 }
        change_variable = { which = ag_num_shell_world_layers value = 1 }
        save_event_target_as = ag_shell_world_base_layer_temp
        if = {
            limit = { OR = { is_colonizable = yes is_under_colonization = yes is_colony = yes } }
            ag_set_shell_world_background_init = yes
            ag_set_shell_world_base = yes
            ag_shell_world_deposit_swap = yes
            ag_set_shell_world_capital = { ag_scope = this }
        }
    }
    else = { event_target:ag_shell_world_base_layer_@this = {
        if = { limit = { exists = event_target:ag_shell_world_surface_layer_@this } clear_global_event_target = ag_shell_world_surface_layer_@this }
        save_event_target_as = ag_shell_world_base_layer_temp
        change_variable = { which = ag_num_shell_world_layers value = 1 }
        clear_global_event_target = ag_shell_world_base_layer_@this
    } }
    if = {
        limit = { has_moon = yes }
        every_moon = {
            root = { prevprev.owner = { add_resource = { minerals = 5000 multiplier = prevprev.trigger:planet_size } } }
            touhou_remove_planet = yes
        }
    }
    solar_system = {
        spawn_planet = {
            name = NAME_touhou_new_shell_world
            class = "pc_ag_shell_world_surface"
            location = prev
            orbit_location = yes
            orbit_distance_offset = 0.001
            size = 1
            has_ring = no
            init_effect = {
                set_planet_entity = { entity = "ag_shell_world_planet_01_entity" }
                set_surveyed = { surveyed = yes surveyor = prevprev.owner }
                set_all_comms_surveyed = yes
                clear_blockers = yes
                clear_planet_modifiers = yes
                set_planet_flag = megastructure
                set_planet_flag = is_megastructure
                set_planet_size = prevprev.trigger:planet_size
                change_planet_size = 1
                set_variable = { which = ag_shell_world_layer value = prevprev.ag_shell_world_layer }
                change_variable = { which = ag_shell_world_layer value = 1 }
                save_event_target_as = ag_shell_world_new_layer
                save_global_event_target_as = ag_shell_world_next_layer_@event_target:ag_shell_world_prev_layer
                set_planet_flag = ag_shell_world_surface_layer
                [[flag] set_planet_flag = $flag$ ]
                set_owner = prevprev.owner
                set_controller = prevprev.owner
                create_pop_group = { species = prevprev.owner.species size = 100 }
                ag_set_shell_world_capital = { ag_scope = prevprev }
                save_event_target_as = $target$
            }
        }
    }
    save_global_event_target_as = ag_shell_world_prev_layer_@event_target:ag_shell_world_new_layer
    if = { limit = { has_planet_flag = ag_shell_world_surface_layer } remove_planet_flag = ag_shell_world_surface_layer }
    if = { limit = { is_planet_class = pc_ag_shell_world_surface } ag_change_pc = { ag_planet_class = pc_ag_shell_world } }
    event_target:ag_shell_world_base_layer_temp = {
        save_global_event_target_as = ag_shell_world_base_layer_@event_target:ag_shell_world_new_layer
    }
    event_target:ag_shell_world_new_layer = {
        save_global_event_target_as = ag_shell_world_surface_layer_@event_target:ag_shell_world_base_layer_temp
    }
    if = {
        limit = { exists = owner }
        owner = {
            [[mega_built_flag] set_country_flag = has_built_or_repaired_megastructure ]
            [[notification]
                if = {
                    limit = { NOT = { has_country_flag = ag_built_first_shell_world } }
                    set_country_flag = ag_built_first_shell_world
                    country_event = { id = toho.4021 }
                }
                else = {
                    create_message = {
                        type = touhou_shell_world_built
                        localization = touhou_shell_world_built_desc
                        days = 5
                        target = event_target:ag_shell_world_new_layer
                    }
                }
            ]
        }
    }
}

touhou_setup_aura_clashing = {
    optimize_memory
    export_trigger_value_to_variable = {
        trigger = monthly_intensity_increase
        variable = monthly_increase
    }
    every_neighbor_system = {
        limit = {
            NOT = { has_star_flag = aura_clashing_on_monthly_check }
            has_psionic_aura = yes
            aura_owner = { NOT = { is_same_value = root.aura_owner } }
            spth_has_psionic_inhibition_station = no
        }

        remove_star_flag = aura_is_clashing_against_@root
        remove_star_flag = aura_clash_winning_against_@root
        root = {
            remove_star_flag = aura_is_clashing_against_@prev
            remove_star_flag = aura_clash_winning_against_@prev
        }

        export_trigger_value_to_variable = {
            trigger = monthly_intensity_increase
            variable = monthly_increase
        }

        if = {
            limit = {
                check_variable = {
                    which = monthly_increase
                    value < root.monthly_increase
                }
            }
            root = { save_event_target_as = winning_system }
            save_event_target_as = losing_system
        }
        else_if = {
            limit = {
                check_variable = {
                    which = monthly_increase
                    value > root.monthly_increase
                }
            }
            save_event_target_as = winning_system
            root = { save_event_target_as = losing_system }
        }
        if = {
            limit = { exists = event_target:losing_system }
            if = {
                limit = {
                    event_target:losing_system = {
                        touhou_aura_clash_can_attacker_spread_in_system = {
                            ATTACKER_SYSTEM = event_target:winning_system
                        }
                    }
                }
                event_target:winning_system = {
                    set_star_flag = aura_is_clashing_against_@event_target:losing_system
                    set_star_flag = aura_clash_winning_against_@event_target:losing_system
                    touhou_trigger_aura_clashing_event = { FROM = event_target:losing_system }
                }
                event_target:losing_system = {
                    set_star_flag = aura_is_clashing_against_@event_target:winning_system
                    touhou_trigger_aura_clashing_event = { FROM = event_target:winning_system }
                }
            }
        }
        else = {
            save_event_target_as = tested_system
            if = { # In case of tie, auras are clashing if any of them can spread
                limit = {
                    OR = {
                        touhou_aura_clash_can_attacker_spread_in_system = {
                            ATTACKER_SYSTEM = root
                        }
                        root = {
                            touhou_aura_clash_can_attacker_spread_in_system = {
                                ATTACKER_SYSTEM = event_target:tested_system
                            }
                        }
                    }
                }
                set_star_flag = aura_is_clashing_against_@root
                touhou_trigger_aura_clashing_event = { FROM = root }
                root = {
                    set_star_flag = aura_is_clashing_against_@event_target:tested_system
                    touhou_trigger_aura_clashing_event = { FROM = event_target:tested_system }
                }
            }
        }
        clear_variable = monthly_increase
    }
    clear_variable = monthly_increase
}
touhou_trigger_aura_clashing_event = {
    optimize_memory
    aura_owner = {
        set_visited = $FROM$
        if = {
            limit = {
                NOT = { has_country_flag = aura_clashing_event_triggered }
            }
            set_country_flag = aura_clashing_event_triggered
            country_event = { id = tutorial.2620 }
            country_event = {
                id = shroud.1510
                scopes = {
                    from = prev
                    fromfrom = $FROM$
                }
            }
        }
    }
}
touhou_try_resolve_aura_clashing = {
    every_neighbor_system = {
        limit = {
            has_star_flag = aura_is_clashing_against_@root
        }
        root = {
            remove_star_flag = aura_is_clashing_against_@prev
            remove_star_flag = aura_clash_winning_against_@prev
        }
        if = {
            limit = { has_star_flag = aura_clash_winning_against_@root }
            export_trigger_value_to_variable = {
                trigger = monthly_intensity_increase
                variable = winning_intensity_candidate
            }
            root = {
                if = {
                    limit = {
                        OR = {
                            NOT = { is_variable_set = highest_winning_intensity }
                            check_variable = {
                                which = highest_winning_intensity
                                value < prev.winning_intensity_candidate
                            }
                        }
                    }
                    set_variable = {
                        which = highest_winning_intensity
                        value = prev.winning_intensity_candidate
                    }
                    prev = { save_event_target_as = current_winner }
                }
            }
            clear_variable = winning_intensity_candidate
            remove_star_flag = aura_clash_winning_against_@root
        }
        remove_star_flag = aura_is_clashing_against_@root
    }
    if = {
        limit = { exists = event_target:current_winner }
        if = {
            limit = { can_aura_spread_in_system = { OWNER = event_target:current_winner.aura_owner } }
            spawn_psionic_aura = { owner = event_target:current_winner.aura_owner }
        }
        clear_variable = highest_winning_intensity
    }
}
