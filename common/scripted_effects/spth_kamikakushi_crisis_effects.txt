spth_spawn_kamikakushi_country = {
    optimize_memory
    if = {

        create_species = {
            name = "NAME_spth_kamikakushi_species"

            immortal = yes

        }
        create_country = {
            name = spth_kamikakushi_country
            type = spth_kamikakushi_country
            species = event_target:spth_kamikakushi_species
            flag = {
                icon = {
                    category = "touhousp"
                    file = "kamikakushi_alt.dds"
                }
                colors = {
                    "purple"
                    "purple"
                    "purple"
                    "null"
                }
                background = {
                    category = "backgrounds"
                    file = "00_solid.dds"
                }
            }
            name_list = touhouchs
            effect = {
                set_graphical_culture = strg_01
                set_country_flag = touhou_mod_country
                set_country_flag = touhou_crisis_country
                set_country_flag = spth_kamikakushi_country
                set_country_flag = protected_from_queen_storm
                add_modifier = { modifier = touhou_immune_from_terrify }
                if = {
                    limit = { NOT = { has_country_flag = has_given_starbase_spth_kamikakushi_designs } }
                    create_ship_design = { design = NAME_starbase_spth_kamikakushi }
                    add_ship_design = last_created_design
                }
                activate_fog_machine = spth_kamikakushi_cloud
                # fog_machine_auto_tracking = yes
                [[debug_spawn_flag] set_country_flag = spth_debug_spawn_flag ]
                save_global_event_target_as = spth_kamikakushi_country
                every_playable_country = {
                    limit = { NOT = { is_same_value = prev } }
                    establish_communications_no_message = prev
                    add_intel = { who = prev amount = 70 }
                }
                every_country = {
                    limit = {
                        OR = {
                            is_country_type = default
                            is_country_type = fallen_empire
                            is_country_type = awakened_fallen_empire
                        }
                        NOT = { has_communications = prev }
                    }
                    establish_communications_no_message = prev
                    add_intel = { who = prev amount = 70 }
                }
                if = {
                    limit = {
                        exists = event_target:spth_makai_defence_system_country
                        event_target:spth_makai_defence_system_country = { has_country_flag = touhou_crisis_country }
                    }
                    set_faction_hostility = {
                        target = event_target:spth_makai_defence_system_country
                        set_friendly = yes
                    }
                }
                country_event = { id = touhou_crisis.19 days = 360 }
            }
        }
    }
}

spth_kamikakushi_init_crisis_leader = {
    optimize_memory
    if = {
        limit = { has_global_flag = has_spth_kamikakushi_species }
        event_target:spth_kamikakushi_country = {
            create_leader = {
                class = commander
                species = event_target:has_spth_kamikakushi_species
                name = random
                skill = 4
                traits = { trait = leader_trait_kamikakushi }
                effect = {
                    set_leader_flag = spth_kamikakushi_leader
                    set_leader_flag = has_gained_level_trait
                    set_leader_flag = legendary_leader
                    add_trait = { trait = subclass_commander_admiral }
                }
            }
            last_created_leader = {
                # roll领袖级别
                [[!force_raity]
                    random_list = {
                        60 = {
                            modifier = { factor = 0 touhou_check_diffiulty_gteq = { value = 3 } }
                            set_leader_flag = leader_kamikakushi_common
                        }
                        30 = {
                            modifier = { factor = 0 touhou_check_diffiulty_lteq = { value = 2 } }
                            set_leader_flag = leader_kamikakushi_rare
                        }
                        15 = {
                            modifier = { factor = 0 touhou_check_diffiulty_lteq = { value = 3 } }
                            set_leader_flag = leader_kamikakushi_master
                        }
                    }
                    reroll_random = yes
                ]
                [[force_raity]
                    set_leader_flag = leader_kamikakushi_$force_raity$
                ]
                # easy以上难度概率roll一个技能
                [[!force_skill]
                    if = {
                        limit = { touhou_check_diffiulty_gteq = { value = 2 } }
                        random_list = {
                            25 = {
                                modifier = { add = -5 touhou_check_diffiulty_gteq = { value = 5 } }
                                add_trait = { trait = leader_trait_kamikakushi_spark }
                            }
                            10 = {
                                modifier = { add = 10 touhou_check_diffiulty_gteq = { value = 5 } }
                                add_trait = { trait = leader_trait_kamikakushi_sakura }
                            }
                            5 = {
                                modifier = { add = 5 touhou_check_diffiulty_gteq = { value = 5 } }
                                modifier = { add = 10 touhou_check_diffiulty_gteq = { value = 5 } }
                                add_trait = { trait = leader_trait_kamikakushi_gap }
                            }
                            40 = {
                                modifier = { add = -20 touhou_check_diffiulty_gteq = { value = 3 } }
                                modifier = { add = -20 touhou_check_diffiulty_gteq = { value = 5 } }
                                # 无事发生
                            }
                        }
                        reroll_random = yes
                    }
                ]
                [[force_skill]
                    add_trait = { trait = $force_skill$ }
                ]
                # 额外技能等级，仅HARD及以上
                [[!force_level]
                    if = {
                        limit = { touhou_check_diffiulty_gteq = { value = 3 } }
                        add_skill = 3
                        if = {
                            limit = { touhou_check_diffiulty_gteq = { value = 4 } }
                            add_skill = 170
                        }
                    }
                ]
                [[force_level]
                    add_skill = $force_level$
                ]
                save_event_target_as = last_created_spth_kamikakushi_leader
            }
            event_target:$fleet$ = {
                assign_leader = event_target:last_created_spth_kamikakushi_leader
            }
        }
    }
}

spth_kamikakushi_spawn_fleets_detail = {
    optimize_memory
    if = {
        limit = { has_global_flag = spth_kamikaushi_crisis_gate_$spth_id$_spawned }
        if = {
            limit = { NOT = { has_global_flag = spth_kamikakushi_gate_$spth_id$_destroyed } }
            event_target:spth_kamikakushi_country = { change_variable = { which = spth_kamikakushi_spawn_interval value = 1 } }
        }
        if = {
            limit = {
                event_target:spth_kamikakushi_country = {
                    if = {
                        limit = { NOT = { has_global_flag = touhou_kamikakushi_extra_21 } }
                        count_owned_fleet = { limit = { has_fleet_flag = spth_kamikakushi_fleet NOT = { has_fleet_flag = spth_kamikakushi_support_fleet } } count <= 48 }
                    } else = { count_owned_fleet = { limit = { has_fleet_flag = spth_kamikakushi_fleet NOT = { has_fleet_flag = spth_kamikakushi_support_fleet } } count <= 60 } }
                }
            }
            random_system = {
                limit = { has_star_flag = spth_kamikaushi_crisis_gate_$spth_id$ }
                random_system_planet = {
                    limit = { has_carrier_flag = spth_kamikaushi_crisis_spawnpoint }
                    save_event_target_as = spth_kamikakushi_spawnpoint
                }
                if = {
                    limit = { NOT = { has_global_flag = spth_kamikakushi_gate_$spth_id$_destroyed } }
                    if = {
                        limit = { NOT = { exists = event_target:spth_kamikakushi_main_fleet_$spth_id$ } }
                        spth_kamikakushi_spawn_main_fleet = { id = $spth_id$ }
                    }
                    event_target:spth_kamikakushi_spawnpoint = {
                        carrier_event = { id = touhou_crisis.111 days = 5 }
                    }
                    event_target:spth_kamikakushi_country = { change_variable = { which = spth_kamikakushi_spawn_interval value = 1 } }
                } else = {
                    event_target:spth_kamikakushi_spawnpoint = {
                        carrier_event = { id = touhou_crisis.122 days = 5 }
                    }
                }
            }
        }
    }
}
