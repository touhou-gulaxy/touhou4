synth_queen_wipe_system = {
    optimize_memory
    # Don't kill ancient system.
    solar_system = { if = {
        limit = {
            touhou_synth_queen_no_wipe_system = no
            NOR = {
                has_star_flag = ag_zeta_crisis_psionic_storm_system
                has_star_flag = ag_zeta_crisis_psionic_singularity_system
            }
        }
        if = { # If we're stealing a player system, warn them
            limit = {
                exists = owner
                owner = {
                    is_country_type = default
                    NOT = { has_country_flag = speech_1	}
                }
            }
            save_event_target_as = synth_queen_lost_system
            owner = { country_event = { id = crisis.8020 days = 1 random = 5 } }
        }
        set_star_flag = synth_queen_system
        # Kill everyone
        every_system_planet = {
            limit = {
                pop_amount > 0
                exists = owner
                owner = { is_primitive = no }
            }
            every_owned_pop_group = { kill_pop_group = { pop_group = this percentage = 1.0 } }
            remove_all_buildings = yes
            touhou_destroy_colony = yes
        }
        # Check fleets
        every_fleet_in_system = {
            # Exclude those owned by the Synth Queen, so she won't accidentally destroy her own battleships
            limit = { owner = { is_synth_queen_country_type = no } }
            if = {
                limit = {
                    NAND = {
                        exists = owner
                        owner = {
                            OR = {
                                is_touhou_crisis_faction = yes
                                is_touhou_other_faction = yes
                                ag_is_psionic_country = yes
                                ag_is_ancient_country = yes
                                is_country_type = ag_ancient_primitive_crisis
                                has_country_flag = ag_invincible_on_system_destroyed_country
                            }
                        }
                    }
                }
                # Destroy all stations
                if = { limit = { is_mobile = no } delete_fleet = this }
                # Ships
                else_if = { #handle ships that should never be killed like that

activate_behemoth_superweapon = {
    root.from = { random_owned_ship = { save_event_target_as = ag_behemoth_superweapon_ship } }
    # Blow up all mining/research stations
    every_system_planet = {
        if = { limit = { exists = orbital_station } orbital_station = { dismantle = yes } }
        if = {
            limit = { ag_is_ancient_ringworld = no }
            # Reset all ecus into relic worlds
            if = {
                limit = { OR = { uses_district_set = city_world uses_district_set = ag_city_planet } }
                relic_world_deposits = yes
                validate_planet_buildings_and_districts = yes
            }
            # Destroy habitats
            else_if = {
                limit = { is_planet_class = pc_habitat }
                if = {
                    limit = { is_colony = yes }
                    remove_all_buildings = yes
                    touhou_destroy_colony = yes
                }
                spawn_habitat_cracker_effect = yes
            }
            # Reset all habitable planets into tomb worlds
            else_if = {
                limit = { ag_is_shell_world = no exists = owner owner = { ag_is_common_country = yes } }
                change_pc = pc_nuked
                set_planet_flag = nuked_planet_anomalies_disabled
                reroll_deposits = yes
                tomb_world_modifier_cleanup = yes
                validate_planet_buildings_and_districts = yes
            }
            else_if = {
                limit = { ag_is_shell_world = yes }
                add_planet_devastation = 80
            }
        }
    }
    # Add 100 devastation to all remaining colonies
    # and kill pops
    every_system_planet = {
        limit = { exists = owner owner = { ag_is_common_country = yes } NOT = { has_modifier = "ag_ancient_shield_modifier" } }
        add_planet_devastation = 100
        random_list = {
            5 = { every_owned_pop_group = { kill_pop_group = { pop_group = this percentage = 0.05 } } }
            45 = { every_owned_pop_group = { kill_pop_group = { pop_group = this percentage = 0.15 } } }
            45 = { every_owned_pop_group = { kill_pop_group = { pop_group = this percentage = 0.20 } } }
            5 = { every_owned_pop_group = { kill_pop_group = { pop_group = this percentage = 0.50 } } }
        }
    }
    # Destroy all fleets in the system
    every_fleet_in_system = {
        limit = { NOR = {
            is_same_value = root
            # except Extradimensionals
            AND = {
                exists = owner
                owner = {
                    OR = {
                        is_touhou_crisis_faction = yes
                        is_touhou_other_faction = yes
                        ag_is_psionic_country = yes
                        ag_is_ancient_country = yes
                        is_country_type = ag_ancient_primitive_crisis
                        has_country_flag = ag_invincible_on_system_destroyed_country
                    }
                }
            }
            # except biggest monsters
            any_owned_ship = { OR = {
                is_ship_size = behemoth_04
                is_ship_size = voidspawn_boss
            } }
        } }
        # Kill every single ships and fire on actions.
        # destroy_fleet = this
        every_owned_ship = {
            limit = { exists = this }
            reroll_random = yes
            random_list = {
                100 = {
                    destroy_ship = this
                    # Ship kill itself.
                    fire_on_action = {
                        on_action = on_ship_disabled
                        scopes = { from = event_target:ag_behemoth_superweapon_ship }
                    }
                    owner = { fire_on_action = {
                        on_action = on_ship_destroyed_victim
                        scopes = {
                            from = root.from.owner
                            fromfrom = prev
                            fromfromfrom = event_target:ag_behemoth_superweapon_ship
                        }
                    } }
                    root.from.owner = { fire_on_action = {
                        on_action = on_ship_destroyed_perp
                        scopes = {
                            from = this
                            fromfrom = event_target:ag_behemoth_superweapon_ship
                            fromfromfrom = prev
                        }
                    } }
                }
            }
        }
        if = {
            limit = { num_ships <= 0 }
            owner = { fire_on_action = {
                on_action = on_fleet_destroyed_victim
                scopes = {
                    from = root.from.owner
                    fromfrom = prev
                    fromfromfrom = root.from
                }
            } }
            root.from.owner = { fire_on_action = {
                on_action = on_fleet_destroyed_perp
                scopes = {
                    from = this
                    fromfrom = root.from
                    fromfromfrom = prev
                }
            } }
        }
    }
    # Deal massive damage to the Elder Voidspawn
    # and disable regen for 1 year
    every_fleet_in_system = {
        limit = { is_ship_size = voidspawn_boss }
        every_owned_ship = { damage_ship = @behemoth_superweapon_damage }
        add_modifier = { modifier = behemoth_superweapon_modifier days = @behemoth_superweapon_duration multiplier = 11 }
    }
}
