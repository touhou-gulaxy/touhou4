synth_queen_wipe_system = {
    optimize_memory
    # Don't kill ancient system.
    solar_system = { if = {
        limit = {
            touhou_synth_queen_no_wipe_system = no
            NOR = {
                has_star_flag = ag_zeta_crisis_psionic_storm_system
                has_star_flag = ag_zeta_crisis_psionic_singularity_system
            }
        }
        if = { # If we're stealing a player system, warn them
            limit = {
                exists = owner
                owner = {
                    is_country_type = default
                    NOT = { has_country_flag = speech_1	}
                }
            }
            save_event_target_as = synth_queen_lost_system
            owner = { country_event = { id = crisis.8020 days = 1 random = 5 } }
        }
        set_star_flag = synth_queen_system
        # Kill everyone
        every_system_planet = {
            limit = {
                pop_amount > 0
                exists = owner
                owner = { is_primitive = no }
            }
            every_owned_pop_group = { kill_pop_group = { pop_group = this percentage = 1.0 } }
            remove_all_buildings = yes
            touhou_destroy_colony = yes
        }
        # Check fleets
        every_fleet_in_system = {
            # Exclude those owned by the Synth Queen, so she won't accidentally destroy her own battleships
            limit = { owner = { is_synth_queen_country_type = no } }
            if = {
                limit = {
                    NAND = {
                        exists = owner
                        owner = {
                            OR = {
                                is_touhou_crisis_faction = yes
                                is_touhou_other_faction = yes
                                ag_is_psionic_country = yes
                                ag_is_ancient_country = yes
                                is_country_type = ag_ancient_primitive_crisis
                                has_country_flag = ag_invincible_on_system_destroyed_country
                            }
                        }
                    }
                }
                # Destroy all stations
                if = { limit = { is_mobile = no } delete_fleet = this }
                # Ships
                else_if = { #handle ships that should never be killed like that

vanilla_overwrite_damage_ship = {
	optimize_memory
	set_variable = { which = dyson_gun_damage_value value = $value$ }
	set_variable = { which = dyson_gun_damage_value_factor value = 1 }
	subtract_variable = { which = dyson_gun_damage_value_factor value = modifier:ship_damage_reduction_mult }
	multiply_variable = { which = dyson_gun_damage_value value = dyson_gun_damage_value_factor }
	clear_variable = dyson_gun_damage_value_factor
	subtract_variable = { which = dyson_gun_damage_value value = modifier:ship_damage_reduction_add }
	if = {
		limit = { has_shield_hp < dyson_gun_damage_value }
		reduce_shield = trigger:has_shield_hp
		subtract_variable = { which = dyson_gun_damage_value value = trigger:has_shield_hp }
		set_variable = { which = dyson_gun_damage_value_neg value = dyson_gun_damage_value }
		multiply_variable = { which = dyson_gun_damage_value_neg value = -1 }
		if = {
			limit = {
				check_variable_arithmetic = {
					which = trigger:has_max_armor_hp
					multiply = trigger:has_armor_percentage
					value < dyson_gun_damage_value
				}
			}
			set_variable = { which = dyson_gun_damage_value_neg value = trigger:has_max_armor_hp }
			multiply_variable = { which = dyson_gun_damage_value_neg value = trigger:has_armor_percentage }
			multiply_variable = { which = dyson_gun_damage_value_neg value = -1 }
			repair_armor_amount = dyson_gun_damage_value_neg
			change_variable = { which = dyson_gun_damage_value value = dyson_gun_damage_value_neg }
			if = {
				limit = { check_variable = { which = dyson_gun_damage_value value > 0 } }
				add_modifier = {
					modifier = vanilla_overwrite_hull_damage days = 2
					multiplier = dyson_gun_damage_value
				}
			}
		} else = {
			repair_armor_amount = dyson_gun_damage_value_neg
		}
		clear_variable = dyson_gun_damage_value_neg
	} else = {
		reduce_shield = dyson_gun_damage_value
	}
	clear_variable = dyson_gun_damage_value
}
