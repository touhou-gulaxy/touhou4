    modifier = {
        add = value:calc_intensity_from_sector_leader
        exists = sector

            is_owned_by = root.aura_owner

                exists = leader

        }

    }
    modifier = {
        add = value:calc_intensity_from_neighbor_auras

        any_system_planet = { is_planet_class = pc_shrouded }

        factor = 0
        system_has_psionic_inhibition_station = yes
        system_has_psionic_burster_station = no

        factor = 0.25
        spth_system_has_psionic_inhibition_station = yes

        desc = AFFECTED_BY_PSIONIC_BURSTER
    }
}
