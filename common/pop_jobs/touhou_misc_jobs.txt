touhou_archivist = {
    category = specialist
    swappable_data = {
        default = {
            building_icon = building_suzunaan
            condition_string = SPECIALIST_JOB_TRIGGER
            icon = touhou_archivist
        }
    }
    possible_pre_triggers = {
        has_owner = yes
        is_being_purged = no
        is_being_assimilated = no
        is_sapient = yes
    }
    possible_precalc = can_fill_specialist_job
    possible = { can_pop_fill_touhou_job = yes }
    can_be_automated = no
    resources = {
        category = planet_touhou_researcher
        produces = {
            physics_research = 6
            engineering_research = 6
            society_research = 6
        }
        produces = {
            trigger = { planet = { is_variable_set = planet_weak_genso_job_produce_faith_modifier } }
            physics_research = 6
            engineering_research = 6
            society_research = 6
            mult = planet.planet_weak_genso_job_produce_faith_modifier
        }
        upkeep = {
            sr_lingli = 2
            consumer_goods = 2
            minor_artifacts = 1
        }
        upkeep = {
            trigger = { planet = { is_variable_set = planet_weak_genso_job_produce_faith_modifier } }
            sr_lingli = 2
            consumer_goods = 2
            minor_artifacts = 1
            mult = planet.planet_weak_genso_job_upkeep_faith_modifier
        }
    }
    inline_script = { script = jobs/touhou_archivist_modifiers }
    tags = { touhou_job touhou_base touhou_research research }
    weight = {
        weight = @spth_specialist_job_weight
        # mult = value:job_weights_research_modifier|JOB|researcher|
        modifier = {
            factor = 0.2
            has_citizenship_rights = no
            NOT = { pop_group_has_trait = trait_mechanical }
        }
        modifier = {
            factor = 2
            has_living_standard = { type = living_standard_academic_privilege }
        }
        modifier = {
            factor = 0.1
            can_take_servant_job = yes
        }
        modifier = {
            factor = 2
            pop_group_has_trait = trait_touhou
        }
    }
}
