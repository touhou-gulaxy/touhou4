	expansion_planner = yes
	show_on_uncolonized = {
		OR = {
			uses_district_set = gensokyo_city
		}
	}

	zone_slots = {
		slot_gensokyo_jinja
	}
	potential = {
        OR = {
            uses_district_set = gensokyo_city
            has_planet_flag = has_shell_world_touhou_gensokyo_city
        }
	}
	
	conversion_ratio = 1
	convert_to = {
		district_city
		district_nexus
		district_hive
	}
	
	resources = {
		category = planet_districts
		cost = {
			minerals = 2000
			sr_lingli = 500
		}
		produces = {
			energy = 10
			minerals = 10
		}
		upkeep = {
			sr_lingli = 20
		}		
	}
	
	planet_modifier = {
		planet_housing_add = 500
		job_spth_story_miner_add = 300
		job_spth_story_technician_add = 300
	}
	inline_script = { script = districts/spth_gensokyo_city_housing }
	triggered_planet_modifier = {
		potential = { has_zone = { zone = spth_gensokyo_city_energy_zone } }
		job_spth_story_miner_add = -300
		job_spth_story_technician_add = 300
	}
	triggered_planet_modifier = {
		potential = { has_zone = { zone = spth_gensokyo_city_minerials_zone } }
		job_spth_story_miner_add = 300
		job_spth_story_technician_add = -300
	}
	triggered_name = { trigger = { always = yes } text = district_gensokyo_jinja_name }
	triggered_flavor_desc = { trigger = { always = yes } text = district_gensokyo_jinja_flavor_desc }

	ai_weight = {
		weight = 0
		modifier = {
			weight = 500
            exists = owner
			owner = {
				has_monthly_income = {
					resource = alloys
					value < 500
				}
				has_monthly_income = {
					resource = minerals
					value > 99
				}
			}
		}
	}
}
