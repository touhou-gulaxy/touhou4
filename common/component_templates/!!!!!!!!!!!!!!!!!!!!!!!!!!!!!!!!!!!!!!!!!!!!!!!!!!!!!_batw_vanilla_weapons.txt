        archaeotech = yes size = titanic type = ARCHAEO_TITAN cooldown = 5.0 size_tag = t tag = energy aoe_range = @batw_perdition_beam_aoe_range
        base_damage = @batw_perdition_beam_fleet_damage fe_color = white fe_size = medium
    }
    target_weights = {
        titan = 3
        battleship = 2

            owner = {
                resource_stockpile_compare = {
                    resource = minor_artifacts
                    value > 250
                }
            }
        }
