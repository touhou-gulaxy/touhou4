weapon_component_template = {
    key = touhou_gunpower_turret_t_weapon
    icon = GFX_ship_part_touhou_gunpower_turret_t_weapon
    custom_tooltip = touhou_gunpower_turret_t_weapon_tt
    inline_script = {
        script = ship_components/touhou_gravity_cannon_turret_weapon_script
        size = titanic life_steal = 0.25 slot_tag = "t_slot weapon_type_touhou_dynamic_damage_ag weapon_type_touhou_gunpower_turret_main_weapon"
        set = "th_gravity_cannon_set_lunar_huge" gfx = th_gravity_cannon_t_sp_weapon_pink
    }
    on_hit = {
        custom_tooltip = touhou_gunpower_turret_t_weapon_on_hit
        hidden_effect = {
            if = {
                limit = { exists = from from = { NOT = { has_ship_flag = touhou_gunpower_turret_t_weapon_cooldown } } }
                from = {
                    if = {
                        limit = { NOT = { is_variable_set = touhou_gunpower_turret_t_weapon_fire_count } }
                        set_variable = { which = touhou_gunpower_turret_t_weapon_fire_count value = 0 }
                    }
                    change_variable = { which = touhou_gunpower_turret_t_weapon_fire_count value = 1 }
                    if = {
                        limit = { check_variable = { which = touhou_gunpower_turret_t_weapon_fire_count value >= 7 } }
                        set_timed_ship_flag = {
                            flag = touhou_gunpower_turret_t_weapon_cooldown
                            days = value:touhou_gunpower_turret_t_weapon_cooldown
                        }
                        add_modifier = {
                            modifier = touhou_gunpower_turret_t_weapon days = 1
                            time_multiplier = value:touhou_gunpower_turret_t_weapon_cooldown|add|1|
                            mult = value:touhou_gunpower_turret_t_weapon_damage_factor
                        }
                    } else = {
                        touhou_weapon_fire_aoe_damage_lunar_gravity_cannon = { size = titanic size_tag = t type = gravity_cannon }
                        prev = {
                            create_ambient_object = {
                                type = th_gravity_cannon_sp_weapon_hit
                                location = this
                                duration = 10
                                use_3d_location = yes
                                entity_offset = { min = 0 max = 0 }
                                entity_offset_angle = { min = 0 max = 0 }
                                entity_offset_height = { min = 0 max = 0 }
                                entity_scale_to_size = no
                                scale = 0.8
                            }
                        }
                    }
                }
            }
        }
    }
    prerequisites = { "tech_touhou_gunpower_turret_gravity_cannon" }
}
