                    limit = { check_variable = { which = th_yuyuko_butterfly_sp_shoot_counter value >= 5140000 } }
                    set_variable = { which = th_yuyuko_butterfly_sp_shoot_counter value = 0 }

                # 160000->16倍, multiply=0.0001
                if = {
                    limit = {
                        check_variable_arithmetic = {
                            which = th_yuyuko_butterfly_sp_shoot_counter
                            modulo = 2
                            value = 1
                        }
                    }
                    # 下次攻击为偶数，基于自身船体值算修正
