    resources = {

        }
        upkeep = { energy = 250 }
    }

    potential = {
        spth_check_gensokyo_auth = yes
        has_technology = tech_gensokyo_ringworld_arkship
    }

    possible = {
        custom_tooltip = {
            fail_text = "requires_inside_border"
            is_inside_border = from
        }
        custom_tooltip = {
            fail_text = spth_gensokyo_ringworld_arkship_already_exists
			from = { check_variable_arithmetic = { which = value:touhou_num_owned_gensokyo_ringworld_arkship value < value:touhou_max_owned_gensokyo_ringworld_arkship } }
        }
    }

    placement_rules = {
        planet_possible = {
            touhou_is_construction_section = yes
        }
    }

    on_build_start = {
        from = {
            if = {
                limit = { NOT = { has_existing_ship_design = gensokyo_ringworld_arkship } }
                create_ship_design = { design = NAME_gensokyo_ringworld_arkship }
                add_ship_design = last_created_design
            }
        }
    }

    on_build_cancel = {
        fromfrom.planet = {
            remove_carrier_flag = has_megastructure
        }
    }

    on_build_complete = {
        fromfrom.planet = {
            set_carrier_flag = has_megastructure
        }
        from = {
            country_event = { id = touhou_mega_misc.5 }
        }
    }

    construction_blocks_and_blocked_by = self_type
    ai_weight = { weight = 10 }
    dismantle_potential = { always = no }

spth_gensokyo_ringworld_arkship_3 = {
    entity = "gensokyo_ringworld_arkship_full_entity"
    construction_entity = "gensokyo_ringworld_arkship_full_entity"
    portrait = "GFX_megastructure_construction_background"
    entity_offset = { x = 0 y = 0 }
    place_entity_on_planet_plane = no
    show_galactic_map_icon = yes
    show_in_outliner = yes
    upgrade_desc = hide
    build_time = 4800
    upgrade_from = { spth_gensokyo_ringworld_arkship_2 }

    resources = {
        category = touhou_megastructures
        cost = {
            alloys = 60000
            sr_lingli = 15000
            sr_fuka = 5000
            sr_dark_matter = 2500
        }
        upkeep = { energy = 250 }
    }

    potential = {
        spth_check_gensokyo_auth = yes
        has_technology = tech_gensokyo_ringworld_arkship
    }

    placement_rules = {}
    on_build_start = {}
    on_build_cancel = {}
    on_build_complete = {
        fromfrom.planet = {
            remove_carrier_flag = has_megastructure
        }
        from = {
            set_country_flag = spth_gensokyo_ringworld_arkship_built
            set_country_flag = has_built_or_repaired_megastructure
            country_event = { id = touhou_mega_misc.8 }
        }
        add_threat = { who = from amount = 50 }
        create_fleet = {
            name = NAME_gensokyo_ringworld_arkship
            settings = {
                spawn_debris = no
                uses_naval_capacity = no
            }
            effect = {
                set_owner = root.from
                set_location = {
                    target = prev
                    distance = 1
                    angle = random
                }
                create_ship = {
                    name = NAME_gensokyo_ringworld_arkship
                    design = NAME_gensokyo_ringworld_arkship
                    prefix = no
                    effect = {
                        set_ship_flag = spth_gensokyo_ringworld_arkship_player
                    }
                }
                set_fleet_stance = aggressive
                set_aggro_range = 300
                set_aggro_range_measure_from = self
                set_fleet_flag = touhou_standard_formation_scale_setted
                set_formation_scale = @touhou_standard_formation_scale
            }
        }
        remove_megastructure = fromfrom
    }

    construction_blocks_and_blocked_by = self_type
    ai_weight = { weight = 10 }
    dismantle_potential = { always = no }
}
