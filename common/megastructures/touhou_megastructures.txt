touhou_system_warper_base = {
    portrait = GFX_megastructure_construction_background
    entity = "construction_platform_entity"
    construction_entity = "construction_platform_entity"
    place_entity_on_planet_plane = no
    entity_offset = { x = -20 y = -20 }
    show_galactic_map_icon = yes
    build_outside_gravity_well = yes
    build_time = 360
    resources = {
        category = touhou_megastructures
        cost = {
            alloys = 5000
            nanites = 2500
            sr_lingli = 5000
        }
        upkeep = {
            energy = 5
        }
    }
    prerequisites = { "tech_touhou_system_warper"  }
    potential = { has_technology = tech_touhou_system_warper is_ai = no }
    possible = {
        hidden_trigger = {
            exists = starbase
        }
        custom_tooltip = {
            fail_text = "requires_inside_border_or_subject"
            OR = {
                is_inside_border = from
                AND = {
                    exists = owner
                    owner = {
                        is_subject = yes
                        overlord = { is_same_value = from }
                    }
                }
            }
        }
        custom_tooltip = {
            fail_text = "requires_surveyed_system" # or is in my subject's territory
            OR = {
                NOT = {
                    any_system_planet = {
                        is_surveyed = {
                            who = prev.from
                            status = no
                        }
                    }
                }
                AND = {
                    exists = owner
                    owner = {
                        is_subject = yes
                        overlord = { is_same_value = from }
                    }
                }
            }
        }
        from = { has_technology = tech_touhou_system_warper }
        custom_tooltip = {
            fail_text = "requires_no_existing_touhou_system_warper"
            from = { check_variable_arithmetic = { which = value:touhou_num_owned_touhou_system_warper value < value:touhou_max_owned_touhou_system_warper } }
        }
    }
    custom_tooltip_requirements = "MEGASTRUCTURE_TOOLTIP_REQUIREMENTS_OUTSIDE_GRAVITY_WELL"
    country_modifier = {
    }
    ai_weight = {
        weight = 0
    }
    dismantle_potential = { always = no }
    on_build_start = {}
    on_build_cancel = {}
    on_build_complete = {
        from = {
            country_event = { id = toho.4212 }
            every_playable_country = {
                limit = { NOT = { is_same_value = prev } }
                country_event = {
                    id = toho.4213
                    scopes = { from = prev fromfrom = prevprev }
                }
            }
        }
    }
    construction_blocks_and_blocked_by = self_type
}
