is_common_district_spth_sp_produce = {
    optimize_memory
    OR = {
        use_touhou_starting_district = no
        AND = {
            spth_has_ag_mod = yes
            check_modifier_value = { modifier = ag_is_ancient_empire_mod_habitable_world value = 0 }
        }
    }
}
spth_has_zone = {
    optimize_memory
    num_zones = {
        type = $zone$
        value > 0
    }
}

    optimize_memory
    OR = {
        is_planet_class = pc_ringworld_tech

        is_planet_class = pc_ringworld_tech_damaged
        is_planet_class = pc_ringworld_seam_damaged
        is_planet_class = pc_ringworld_shielded
    }
}
