check_aura_suppressors = {
    optimize_memory
    NOT = { any_ship_in_system = { check_modifier_value = { modifier = "is_psionic_burster" value != 0 } } }
    OR = {
        any_system_colony = {
            OR = {
                has_active_building = building_psionic_suppressor
                has_active_building = building_ancient_ward_2
            }
        }
        has_megastructure = shroud_seal
        check_variable_arithmetic = { which = value:count_neighbor_shroud_seals value > 0 }

        any_ship_in_system = { check_modifier_value = { modifier = "is_psionic_inhibitor" value != 0 } }
        tc_check_aura_suppressors = yes
    }
}

        has_technology = tech_sapient_ai
        has_technology = tech_trait_th_alice_doll
        any_relation = {
            has_communications = prev
            OR = {
                has_technology = tech_synthetic_workers
                has_technology = tech_sapient_ai
                is_machine_empire = yes
                is_individual_machine = yes
            }
            OR = {
                is_country_type = default
                has_ai_personality_behaviour = robot_exploiter
                has_ai_personality_behaviour = robot_liberator
            }
        }
    }
}
can_fill_specialist_job_trigger = {
    optimize_memory
    custom_tooltip = SPECIALIST_JOB_TRIGGER
    hidden_trigger = {
        NOT = { has_ethic = ethic_gestalt_consciousness }
        exists = owner
        OR = {
            is_enslaved = no
            has_slavery_type = { type = slavery_indentured }
        }
        is_being_purged = no
        is_being_assimilated = no
        NOT = { has_trait = trait_syncretic_proles }
        # Rule out Traits for servitude & lack of free will
        can_think = yes
        NOT = {	has_pop_group_flag = cant_work }
        # Rule out Disconnected Drones
        has_disconnected_drone_citizenship_type = no
        OR = {
            NOT = { has_trait = trait_mechanical }
            has_trait = trait_th_alice_doll
            owner = { has_technology = tech_droid_workers }
        }
        NOT = {
            has_citizenship_type = {
                type = citizenship_organic_trophy
                country = owner
            }
        }
        if = {
            limit = {
                divinity_right_to_work_job_check_trigger_exempt = no
            }
            divinity_right_to_work_job_check_trigger_specialist = yes
        }
    }
}
complex_specialist_job_check_trigger = {
    optimize_memory
    hidden_trigger = {
        exists = owner
        OR = {
            NOT = { has_trait = trait_mechanical }
            has_trait = trait_th_alice_doll
            AND = {
                owner = { has_technology = tech_droid_workers }
                owner = { NOT = { has_policy_flag = ai_outlawed } }
            }
        }
    }
}
can_be_leader = {
    optimize_memory
    OR = {
        has_citizenship_rights = yes
        has_citizenship_type = { type = citizenship_shroud_touched country = root }
    }
    # Rule out Traits for servitude & lack of free will
    can_think = yes
    OR = {
        is_robot_pop_group = no
        if = {
            limit = { has_trait = trait_th_alice_doll }
            root = {
                OR = {
                    has_authority = auth_gensokyo
                    is_machine_empire = yes
                    is_mechanical_empire = yes
                    is_individual_machine = yes
                    has_country_flag = synthetic_fertility_no_bodies_left
                }
            }
        } else = {
            root = {
                OR = {
                    has_technology = tech_synthetic_leaders
                    is_machine_empire = yes
                    is_mechanical_empire = yes
                    is_individual_machine = yes
                    has_country_flag = synthetic_fertility_no_bodies_left
                }
            }
        }
    }
    OR = {
        is_same_species = root
        root = {
            NAND = {
                has_origin = origin_necrophage
                has_trait = trait_necrophage
            }
        }
    }
}
