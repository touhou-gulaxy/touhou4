touhou_is_dyson_gun_counterattack_system = {
	optimize_memory
	OR = {
        space_owner = {
            OR = {
                is_touhou_crisis_faction = yes
                is_touhou_other_faction = yes
            }
        }
		any_system_planet_colony = {
			owner? = {
                OR = {
                    is_touhou_crisis_faction = yes
                    is_touhou_other_faction = yes
                }
			}
		}
		any_ship_in_system = {
			owner? = {
                OR = {
                    is_touhou_crisis_faction = yes
                    is_touhou_other_faction = yes
                }
			}
		}
	}
}
