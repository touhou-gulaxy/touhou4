can_trigger_touhou_ideology_1550_event = {
	NOT = { has_country_flag = touhou.1550_fired }

can_trigger_touhou_ideology_1560_event = {
	NOT = { has_country_flag = touhou.1560_fired }
	num_active_tenets > 0
}

can_trigger_touhou_ideology_1570_event = {
	NOT = { has_country_flag = touhou.1570_fired }
	any_owned_colony = {
		free_building_slots > 0
		has_memorial = no
	}
	resource_stockpile_compare = {
		resource = energy
		value >= 500
	}
}

can_trigger_touhou_ideology_1660_event = {
	NOT = { has_country_flag = touhou.1660_fired }

			value >= 3000
		}
		ideological_strength >= 100
	}
}

can_trigger_touhou_ideology_1670_event = {
	NOT = { has_country_flag = touhou.1670_fired }
	any_owned_leader = {
		is_ruler = no
		OR = {
			exists = planet
			leader_class = official
		}
		NOT = { has_trait = leader_trait_maimed }
	}
	resource_stockpile_compare = {
		resource = unity
		value >= 1000
	}
}

can_trigger_touhou_ideology_1680_event = {
	NOT = { has_country_flag = touhou.1680_fired }
	resource_stockpile_compare = {
		resource = unity
		value >= 1000
	}
	OR = {
		any_owned_colony = {
			num_assigned_jobs = {
				job = politician
				value >= 100
			}
		}
		any_owned_colony = {
			num_assigned_jobs = {
				job = bureaucrat
				value >= 100
			}
		}
	}
}

can_trigger_touhou_ideology_1690_event = {
	NOT = { has_country_flag = touhou.1690_fired }
	any_owned_leader = {
		is_ruler = no
		OR = {
			exists = planet
			is_councilor = yes
		}
		NOT = { has_trait = leader_trait_dissident }
	}
}

can_trigger_touhou_ideology_1700_event = {
	NOT = { has_country_flag = touhou.1700_fired }
	any_owned_planet = {
		num_buildings = { type = any value > 1 }
		any_owned_pop_group = {
			pop_group_size >= 500
		}
		OR = {
			num_assigned_jobs = {
				job = enforcer
				value > 0
			}
			owner = {
				ideological_strength >= 100
			}
		}
	}
}

can_trigger_touhou_ideology_1710_event = {
	NOT = { has_country_flag = touhou.1710_fired }
	any_owned_leader = {
		is_ruler = no
		OR = {
			exists = planet
			leader_class = official
		}
	}
}

can_trigger_touhou_ideology_1720_event = {
	NOT = { has_country_flag = touhou.1720_fired }
	count_owned_fleet = {
		limit = {
			exists = leader
			is_ship_class = shipclass_military
		}
		count > 1
	}
	resource_stockpile_compare = {
		resource = unity
		value >= 1000
	}
}
