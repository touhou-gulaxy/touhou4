ship_from_touhou_country = {
    optimize_memory
    exists = from
    from = {
        has_authority = auth_gensokyo
        [[flag] has_country_flag = $flag$ ]
        [[ap] has_touhou_ascension_tier = { tier = $ap$ } ]
        [[origin] has_origin = $origin$ ]
        [[not_origin] NOT = { has_origin = $not_origin$ } ]
    }
}
