touhou_gateway = {
    name = touhou_gateway
    is_pathfind = yes
    uses_action = no
    always_active = yes
    requires_exploration = no
    windup_time = 0
    winddown_time = 0
    ftl_multiplier = 0
    icon_frame = 42
    remember_instances = no
    galactic_map_tooltip_header = touhou_gateway
    galactic_map_tooltip_desc = touhou_gateway_desc
    connection_type = any_other
    can_bypass_be_locked = no
    animation_time = 1
    ftl_ship_effect = "gateway_ftl_ship_effect_entity"
    ftl_bypass_effect = "gateway_ftl_bypass_effect_entity"

    country_can_use = {
        custom_tooltip = {
            fail_text = touhou_gateway_access_denied
            OR = {
                AND = {
                    NOT = { exists = event_target:touhou_gateway_network_controller }
                    OR = {
                        NOT = { exists = from }
                        from = {
                            NOR = {
                                has_closed_borders = root
                                is_hostile_to = root
                            }
                        }
                    }
                }
                AND = {
                    exists = event_target:touhou_gateway_network_controller
                    is_same_value = event_target:touhou_gateway_network_controller
                }
                AND = {
                    exists = event_target:touhou_gateway_network_controller
                    event_target:touhou_gateway_network_controller = {
                        NOR = {
                            has_country_flag = touhou_gateway_network_policy_whitelist
                            has_country_flag = touhou_gateway_network_policy_blacklist
                            is_same_value = root
                            has_closed_borders = root
                            is_hostile = root
                        }
                    }
                }
                AND = {
                    exists = event_target:touhou_gateway_network_controller
                    event_target:touhou_gateway_network_controller = {
                        has_country_flag = touhou_gateway_network_policy_blacklist
                        NOT = { has_country_flag = touhou_gateway_network_disallow_@root }
                    }
                }
                AND = {
                    exists = event_target:touhou_gateway_network_controller
                    event_target:touhou_gateway_network_controller = {
                        has_country_flag = touhou_gateway_network_policy_whitelist
                        has_country_flag = touhou_gateway_network_allow_@root
                    }
                }
            }
        }
    }
    on_action = on_entering_touhou_gateway
}

                OR = {
                    is_country_type = lunar_capital_remain_fleet

                    }
                }
            }
            always = yes
        }
        else = {
            NOT = { has_global_flag = lunar_jumpdrive_gateway_blocked }
        }
	}
}
