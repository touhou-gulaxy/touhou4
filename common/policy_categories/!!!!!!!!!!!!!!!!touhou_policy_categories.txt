# Policy categories: cosmetic groupings for the Policies UI (collapsible headers on the Policies tab
# and the per-category policy popup). They carry no gameplay effect.
#
# Display order is alphabetical by localized category name, except "uncategorized" which is always
# sorted last. Policies are placed in a category via "category = <key>" in their definition, or in
# several at once via "category = { <key> <key> }" (the policy then appears under each listed
# category). Policies with no category (including unmaintained mods) fall back to "uncategorized".
#
# Each category may set "icon = <sprite_key>". The icon is drawn as an inline text-icon before the
# category name in both the collapsible header and the per-category popup. IMPORTANT: keep the icon
# in this "icon" field, never baked into the localized name - the UI sorts categories alphabetically
# by their localized name, so an inline icon token in the loc string would sort on the sprite key
# instead of the visible word.
touhou = { icon = "touhou" }
