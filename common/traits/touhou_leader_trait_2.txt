        species_leader_exp_gain = 0.15

        potential = { leader_class = commander }

        ship_spth_armor_efficiency = 0.075

    triggered_modifier = {
        potential = { leader_class = scientist }
        ship_archaeological_site_clues_add = 1

        ship_anomaly_research_speed_mult = 0.15

    triggered_planet_modifier = {
        potential = { leader_class = official }
        planet_jobs_upkeep_mult = -0.05
        planet_structures_cost_mult = -0.05

    custom_subtitle = leader_trait_touhou_common_subtitle
