# Surging events trigger
country_event = {
	id = touhou.1500
	is_triggered_only = yes
	hide_window = yes

	trigger = { has_force_of_will_dlc = yes }

	immediate = {
		if = {
			limit = { is_gestalt = yes }
			fire_on_action = {
				on_action = on_touhou_ideology_momentum_positive_gestalt
				scopes = { from = from }
			}
			break = yes
		}

		if = {
			limit = {
				can_trigger_touhou_ideology_momentum_surging_events = yes
				has_any_touhou_ideology_momentum_very_positive_events_fired = yes
			}
			fire_on_action = {
				on_action = on_touhou_ideology_momentum_surging
				scopes = { from = from }
			}
			break = yes
		}

		if = {
			limit = {
				can_trigger_touhou_ideology_momentum_very_positive_events = yes
				has_any_touhou_ideology_momentum_positive_events_fired = yes
			}
			fire_on_action = {
				on_action = on_touhou_ideology_momentum_very_positive
				scopes = { from = from }
			}
			break = yes
		}

		if = {
			limit = { can_trigger_touhou_ideology_momentum_positive_events = yes }
			fire_on_action = {
				on_action = on_touhou_ideology_momentum_positive
				scopes = { from = from }
			}
		}
	}
}

	id = touhou.1570
	title = touhou.1570.name
	desc = touhou.1570.desc
	picture = GFX_evt_spth_gensokyo_city
	show_sound = event_construction
	is_triggered_only = yes

	trigger = {
		has_force_of_will_dlc = yes
		can_trigger_touhou_ideology_1570_event = yes
	}

	immediate = {
		set_country_flag = touhou.1570_fired
		set_timed_country_flag = {
			flag = recently_triggered_momentum_event
			days = 3600
		}
		ordered_owned_colony = {
			limit = {
				free_building_slots > 0
				has_memorial = no
			}
			position = 0
			order_by = trigger:pop_amount
			save_event_target_as = selected_planet
			add_memorial_1 = yes
		}
	}

	after = {
		if = {
			limit = { is_memorialist_empire = yes }
			custom_tooltip = building_galactic_memorial_tt
		}
		else_if = {
			limit = { is_megacorp = yes }
			custom_tooltip = building_corporate_monument_tt
		}
		else = {
			custom_tooltip = building_autochthon_monument_tt
		}
		if = {
			limit = {
				NOT = { has_technology = tech_cultural_heritage }
				can_research_technology = tech_cultural_heritage
			}
			give_technology = {
				tech = tech_cultural_heritage
				message = no
			}
		}
	}

	option = {
		name = touhou.1570.a
		allow = {
			resource_stockpile_compare = {
				resource = energy
				value >= 500
			}
		}
		add_resource = { energy = -500 }
		add_modifier = {
			modifier = touhou_ideology_human_village_recognition
			years = 10
		}
	}

	option = {
		name = touhou.1570.b
		default_hide_option = yes
	}
}
