#露米娅舰队
country_event = {
	id = th_boss.10
	title = "th_boss.10.name"
	desc = "th_boss.10.desc"
	picture = GFX_evt_th_wwj
	show_sound = event_ship_thrusters
	location = event_target:cultist_ship
    diplomatic = yes
    custom_gui = spth_event_large_window
    custom_gui_option = spth_event_large_ui_option
    picture_event_data = { room = spth_fleet_evil_room }
	is_triggered_only = yes
	trigger = {
		is_ai = no
	}
	immediate = {
		set_country_flag = boss_event_1
		save_event_target_as = owner_country
		capital_scope = {
			solar_system = {
				random_system_planet = {
					limit = {
						is_capital = no
						is_asteroid = no
						is_star = no
						is_moon = no
						NOT = {
							is_planet_class = pc_gas_giant
						}
					}
					save_event_target_as = hijacked_fleet_planet
				}
                every_system_ambient_object = {
                    limit = { is_ambient_object_type = touhou_disabled_super_flagship }
                    destroy_ambient_object = this
                }
			}
		}
		owner_species = {
			save_event_target_as = owner_species
		}
		create_country = {
			name = "NAME_lumiya"
			species = event_target:owner_species
			type = faction
			name_list = ROOT
			auto_delete = no
			effect = {
				establish_communications_no_message = root
				establish_contact = {
					who = root
					location = root.capital_scope
				}
				create_leader = {
					class = scientist
					species = owner_species
					name = "NAME_lumiya"
					skill = 10
					traits = {
						trait = leader_trait_lumiya_boss
						trait = random_common
					}
                    effect = {
                        change_leader_portrait = spth_por_lumiya
                    }
				}
				create_fleet = {
					name = "NAME_lumiya"
					effect = {
						set_owner = prev
						create_ship = {
							name = "NAME_lumiya"
							design = "NAME_th_boss_lumiya_new"
							graphical_culture = strg_01
							effect = {
								set_ship_flag = th_boss_ship1
								set_disable_at_health = 0.01
								save_event_target_as = th_boss_ship1
							}
						}
						set_location = event_target:hijacked_fleet_planet
						#save_event_target_as = 
						set_fleet_stance = aggressive
						set_aggro_range_measure_from = return_point
						set_aggro_range = 50
						assign_leader = last_created_leader
						queue_actions = {
							repeat = {
								wait = {
									duration = 100
								}
							}
						}
					}
				}
				add_ship_design = last_created_design
				save_event_target_as = boss_event_1
				set_name = random
				set_country_flag = lumiya_boss_of@root
			}
		}
	}
	option = {
		name = "th_boss.10.a"
	}
}
