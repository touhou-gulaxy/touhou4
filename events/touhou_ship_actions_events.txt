fleet_event = {
    id = touhou_ship_actions.3
    title = touhou_ship_actions.3.name
    desc = touhou_ship_actions.3.desc
    is_triggered_only = yes
    diplomatic = yes
    custom_gui = spth_system_selector_window
    custom_gui_option = spth_system_selector_option
    immediate = {
        owner = { set_country_flag = has_spth_system_selector_opened }
        save_global_event_target_as = spth_system_selector_reimu_fleet_@owner
    }
    option = {
        name = "spth_system_selector_target_confirm"
        default_hide_option = yes
        custom_gui = spth_system_selector_confirm_reimu_skill
    }
    option = {
        name = "spth_system_selector_target_abort"
        hidden_effect = { remove_fleet_flag = touhou_ship_action_leader_reimu_cooldown }
        fleet_event = { id = touhou_ship_actions.2 }
    }
    after = {
        hidden_effect = {
            owner = { remove_country_flag = has_spth_system_selector_opened }
            clear_global_event_target = spth_system_selector_reimu_fleet_@owner
        }
    }
}
