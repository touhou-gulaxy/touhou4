country_event = {
	id = th_weapon_up.1
	hide_window = yes

	is_triggered_only = yes
	trigger = {
		has_authority = auth_gensokyo
		FROMFROM = {
			exists = leader
			leader = {
				leader_class = commander
				has_leader_flag = spth_sp_leader
			}
		}
		NOT = { has_modifier = spth_p_max }
	}
	immediate = {
		change_variable = {
			which = spth_number_p
			value = 2
		}
		if = {
			limit = {
				check_variable = {
					which = spth_number_p
					value = 8
				}
			}

			country_event = {
				id = th_weapon_up.2
			}
		}
		
		else_if = {
			limit = {
				check_variable = {
					which = spth_number_p
					value = 16
				}
			}

			country_event = {
				id = th_weapon_up.2
			}
		}
		else_if = {
			limit = {
				check_variable = {
					which = spth_number_p
					value = 32
				}
			}

			country_event = {
				id = th_weapon_up.2
			}
		}
		else_if = {
			limit = {
				check_variable = {
					which = spth_number_p
					value = 48
				}
			}

			country_event = {
				id = th_weapon_up.2
			}
		}
		else_if = {
			limit = {
				check_variable = {
					which = spth_number_p
					value = 64
				}
			}

			country_event = {
				id = th_weapon_up.2
			}
		}
		else_if = {
			limit = {
				check_variable = {
					which = spth_number_p
					value = 80
				}
			}

			country_event = {
				id = th_weapon_up.2
			}
		}
		else_if = {
			limit = {
				check_variable = {
					which = spth_number_p
					value = 96
				}
			}

			country_event = {
				id = th_weapon_up.2
			}
		}
		
		else_if = {
			limit = {
				check_variable = {
					which = spth_number_p
					value = 112
				}
			}

			country_event = {
				id = th_weapon_up.2
			}
		}
		else = {
			if = {
				limit = { check_variable_arithmetic = { which = spth_number_p modulo = 16 value = 0 } }
				if = {
					limit = { has_modifier = spth_p_max }
					remove_modifier = spth_p_max
				}
				if = {
					limit = { has_modifier = spth_p_max_ex }
					remove_modifier = spth_p_max_ex
				}
				add_modifier = {
					modifier = spth_p_max_ex
					multiplier = value:spth_p_max_ex_factor
				}
			}
		}
		
	}
}

	id = th_weapon_up.103

	is_triggered_only = yes

		from = { is_country_type = touhou_ac_research_country }
		NOT = { has_country_flag = th_weapon_up_lock }
		OR = {
			has_country_flag = th_weapon_up_start
			NOT = { has_country_flag = th_weapon_up_edict_time }
		}
		is_ai = no has_ascension_perk = ap_hm6
	}
	immediate = {
		set_country_flag = th_weapon_up_lock
		country_event = { id = th_weapon_up.10 }
	}
