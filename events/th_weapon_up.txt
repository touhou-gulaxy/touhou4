			}
		}

	}

# check monthly, 80% chance to +1, 20% to +0.5, give tech when progress at 10.
country_event = {
	id = th_weapon_up.200
	hide_window = yes
	is_triggered_only = yes
	trigger = { has_country_flag = th_weapon_up_research_begin is_ai = no }
	immediate = {
		if = {
			limit = { NOT = { is_variable_set = th_weapon_up_research_progress } }
			set_variable = { which = th_weapon_up_research_progress value = 0 }
		}
		reroll_random = yes
		random_list = {
			20 = { change_variable = { which = th_weapon_up_research_progress value = 0.5 } }
			8 = {
				modifier = { exists = ruler mult = ruler.trigger:has_total_skill }
				change_variable = { which = th_weapon_up_research_progress value = 1 }
			}
			1 = {
				modifier = { exists = ruler mult = ruler.trigger:has_total_skill }
				change_variable = { which = th_weapon_up_research_progress value = 2 }
			}
		}
		if = {
			limit = { check_variable = { which = th_weapon_up_research_progress value >= 10 } }
			set_variable = { which = th_weapon_up_research_progress value = 0 }
			create_message = {
				type = th_weapon_up_research_finished
				days = 30 target = root recipient = root
				localization = MESSAGE_th_weapon_up_research_finished_DESC
			}
			if = {
				limit = { check_variable = { which = th_weapon_up_research_type value = @th_weapon_up_research_gungnir } }
				give_technology = { tech = tech_repeatable_improved_gungnir }
			}
			else_if = {
				limit = { check_variable = { which = th_weapon_up_research_type value = @th_weapon_up_research_knife } }
				give_technology = { tech = tech_repeatable_improved_knife }
			}
			else_if = {
				limit = { check_variable = { which = th_weapon_up_research_type value = @th_weapon_up_research_xianzhe } }
				give_technology = { tech = tech_repeatable_improved_xianzhe }
			}
			else_if = {
				limit = { check_variable = { which = th_weapon_up_research_type value = @th_weapon_up_research_spark } }
				give_technology = { tech = tech_repeatable_improved_spark }
			}
			else_if = {
				limit = { check_variable = { which = th_weapon_up_research_type value = @th_weapon_up_research_needle } }
				give_technology = { tech = tech_repeatable_improved_needle }
			}
			else_if = {
				limit = { check_variable = { which = th_weapon_up_research_type value = @th_weapon_up_research_reimu } }
				give_technology = { tech = tech_repeatable_improved_reimu }
			}
			else_if = {
				limit = { check_variable = { which = th_weapon_up_research_type value = @th_weapon_up_research_sanae } }
				give_technology = { tech = tech_repeatable_improved_sanae }
			}
			else_if = {
				limit = { check_variable = { which = th_weapon_up_research_type value = @th_weapon_up_research_laevatain } }
				give_technology = { tech = tech_repeatable_improved_laevatain }
			}
		}
