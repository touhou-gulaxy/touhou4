    id = th_AI_plus.1
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        owner = {
            is_ai = yes
            has_authority = auth_gensokyo
        }
    }
    immediate = {
        # 首府升级
        if = {
            limit = { has_building = building_colony_shelter }
            remove_building = building_colony_shelter
            add_building = building_capital
        }
        # 决议
        if = {
            limit = { NOT = { has_deposit = d_hakurei_barrier } }
            select_decision = { name = decision_establish_barrier }
        }
        # 龙神像
        if = {
            limit = { NOT = { has_building = building_spth_cloner } }
            add_building = building_spth_cloner
        }
        if = {
            limit = { NOT = { has_building = building_mls_1 } }
            add_building = building_mls_1
            create_pop_group = {
                species = owner_species
                size = 500
            }
        }
    }
}

country_event = {
    id = th_AI_plus.4
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        NOT = { end_game_years_passed >= 0 }
        NOT = { has_global_flag = spth_AI_NOT_plus }
        is_ai = yes has_authority = auth_gensokyo
        mid_game_years_passed >= 0
    }
    immediate = {
        if = {
            limit = {
                NOT = { has_technology = tech_th_battle_cruiser }
                can_research_technology = tech_th_battle_cruiser
            }
            give_technology = { tech = tech_th_battle_cruiser }
            create_ship_design = { design = NAME_touhou_ai_design_ship_carrier }
            add_ship_design = last_created_design
            create_ship_design = { design = NAME_touhou_ai_design_ship_missile }
            add_ship_design = last_created_design
            create_ship_design = { design = NAME_touhou_ai_design_ship_gunpower_scarlet }
            add_ship_design = last_created_design
            create_ship_design = { design = NAME_touhou_ai_design_ship_gunpower_marisa }
            add_ship_design = last_created_design
        }
        if = {
            limit = { ruler = { NOT = { has_trait = leader_trait_ruler_spth_AI_ruler } } }
            ruler = { add_trait = leader_trait_ruler_spth_AI_ruler }
        }
        if = {
            limit = { count_owned_fleet = { count <= 5 } }
            if = {
                limit = { has_global_flag = spth_AI_plus_extra }
                create_fleet = {
                    set_take_point = yes
                    effect = {
                        set_owner = root
                        set_variable = { which = th_ai_ship_count value = root.value:touhou_country_command_limit }
                        divide_variable = { which = th_ai_ship_count value = 16 }
                        floor_variable = th_ai_ship_count
                        create_ship = { name = random random_existing_design = TH_battle_cruiser effect = { set_ship_flag = disable_th_ai_plus_delete } }
                        clear_variable = th_ai_ship_count
                        set_location = root.capital_scope
                        set_fleet_bombardment_stance = spth_kamikakushi
                    }
                }
            } else = {
                create_fleet = {
                    set_take_point = yes
                    effect = {
                        set_owner = root
                        set_variable = { which = th_ai_ship_count value = root.value:touhou_country_command_limit }
                        divide_variable = { which = th_ai_ship_count value = 8 }
                        floor_variable = th_ai_ship_count
                        random_list = {
                            50 = { create_ship = { name = random design = NAME_spth_battle1 effect = { set_ship_flag = disable_th_ai_plus_delete } } }
                            50 = { create_ship = { name = random design = NAME_spth_battle2 effect = { set_ship_flag = disable_th_ai_plus_delete } } }
                        }
                        clear_variable = th_ai_ship_count
                        set_location = root.capital_scope
                        set_fleet_bombardment_stance = spth_kamikakushi
                    }
                }
            }
        }
    }
}

country_event = {
    id = th_AI_plus.5
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        NOT = { has_global_flag = spth_AI_NOT_plus }
        is_ai = yes has_authority = auth_gensokyo
        end_game_years_passed >= 0
    }
    immediate = {
        if = {
            limit = { ruler = { NOT = { has_trait = leader_trait_ruler_spth_AI_ruler } } }
            ruler = { add_trait = leader_trait_ruler_spth_AI_ruler }
        }
        if = {
            limit = { has_global_flag = spth_AI_plus_extra }
            random_owned_fleet = {
                limit = {
                    is_ship_class = shipclass_military is_cloaked = no
                    is_in_combat = no NOT = { has_fleet_flag = disable_th_ai_plus_delete }
                    is_disabled = no is_upgrading = no
                }
                set_fleet_bombardment_stance = spth_kamikakushi
                set_variable = { which = th_ai_convert_count value = 0 }
                create_ship = { name = random random_existing_design = TH_battle_cruiser effect = { set_ship_flag = disable_th_ai_plus_delete } }
                every_owned_ship = {
                    limit = {
                        is_designable = yes is_disabled = no
                        OR = {
                            ship_size_multiplier < 8
                            is_touhou_ship = no
                        }
                        NOT = { has_ship_flag = disable_th_ai_plus_delete }
                    }
                    delete_ship = this prev = { change_variable = { which = th_ai_convert_count value = prev.trigger:ship_size_multiplier } }
                }
                divide_variable = { which = th_ai_convert_count value = 16 }
                while = {
                    count = th_ai_convert_count
                    create_ship = { name = random random_existing_design = TH_battle_cruiser effect = { set_ship_flag = disable_th_ai_plus_delete } }
                }
            }
            if = {
                limit = { count_owned_fleet = { count <= 10 } }
                create_fleet = {
                    set_take_point = yes
                    effect = {
                        set_owner = root
                        set_variable = { which = th_ai_ship_count value = root.value:touhou_country_command_limit }
                        divide_variable = { which = th_ai_ship_count value = 16 }
                        floor_variable = th_ai_ship_count
                        create_ship = { name = random random_existing_design = TH_battle_cruiser effect = { set_ship_flag = disable_th_ai_plus_delete } }
                        clear_variable = th_ai_ship_count
                        set_location = root.capital_scope
                        set_fleet_bombardment_stance = spth_kamikakushi
                    }
                }
            }
        }
    }
}
