# spawn kedama
event = {
    id = touhou_kedama.1
    is_triggered_only = yes
    hide_window = yes
    immediate = {
        touhou_create_kedama_country = yes
    }
}

country_event = {
    id = touhou_kedama.2
    is_triggered_only = yes
    hide_window = yes
    trigger = { NOT = { has_global_flag = touhou_kedama_respawn_cooldown } }
    immediate = {
        set_timed_global_flag = { flag = touhou_kedama_respawn_cooldown days = 3600 }
        event_target:kedama_home_system = {
            random_system_planet = {
                limit = { has_planet_flag = kedama }
                save_event_target_as = kedama_spawn_location
            }
        }
        touhou_create_kedama_fleet_large = yes
        touhou_create_kedama_fleet_mid = yes
        touhou_create_kedama_fleet_small = yes
        touhou_create_kedama_fleet_small = yes
    }
}
event = {
    id = touhou_kedama.3
    is_triggered_only = yes
    hide_window = yes
    trigger = { NOT = { has_global_flag = touhou_kedama_respawn_cooldown } }
    immediate = {
        event_target:kedama_home_system = {
            random_system_planet = {
                limit = { has_planet_flag = kedama }
                save_global_event_target_as = kedama_spawn_location
            }
        }
        if = {
            limit = { event_target:kedama_country = { count_owned_fleet = { limit = { has_fleet_flag = kedama_large } count < 3 } } }
            touhou_create_kedama_fleet_large = yes
        }
        if = {
            limit = { event_target:kedama_country = { count_owned_fleet = { limit = { has_fleet_flag = kedama_mid } count < 6 } } }
            touhou_create_kedama_fleet_mid = yes
        }
        if = {
            limit = { event_target:kedama_country = { count_owned_fleet = { limit = { has_fleet_flag = kedama_small } count < 12 } } }
            touhou_create_kedama_fleet_small = yes
        }
    }
}
