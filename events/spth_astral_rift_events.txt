    id = spth_astral_rift.118
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        has_country_flag = th_tenshi_get
        has_authority = auth_gensokyo is_ai = no
        mid_game_years_passed >= 5
        NOT = { has_country_flag = spth_astral_rift_118_fired }
    }
    immediate = {
        reroll_random = yes
        random_list = {
            10 = {
                modifier = { add = 5 mid_game_years_passed >= 10 }
                modifier = { add = 5 mid_game_years_passed >= 15 }
                modifier = { add = 5 mid_game_years_passed >= 20 }
                modifier = { add = 5 mid_game_years_passed >= 25 }
                modifier = { add = 10 end_game_years_passed >= 5 }
                modifier = { add = 10 end_game_years_passed >= 10 }
                set_country_flag = spth_astral_rift_118_fired
                country_event = { id = spth_astral_rift.119 days = 30 random = 90 }
            }
            90 = {
                modifier = { subtract = 5 mid_game_years_passed >= 10 }
                modifier = { subtract = 5 mid_game_years_passed >= 15 }
                modifier = { subtract = 5 mid_game_years_passed >= 20 }
                modifier = { subtract = 5 mid_game_years_passed >= 25 }
                modifier = { subtract = 10 end_game_years_passed >= 5 }
                modifier = { subtract = 10 end_game_years_passed >= 10 }
                modifier = { set = 0 years_passed >= 190 }
                log = "oops, random failed to trigger spth_astral_rift.119!"
            }
        }
    }
}
