    id = lunar_capital_psi.9
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        has_global_flag = spth_lunar_capital_psi_crisis_begin
        NOT = { has_global_flag = spth_lunar_capital_psi_crisis_ended }
        has_global_flag = touhou_lunar_psi_extra_11
        exists = fromfrom fromfrom = { exists = solar_system }
        OR = { is_country_type = default is_ai = yes }
        fromfromfrom = { has_fleet_flag = spth_lunar_psi_fleet }
        from = { is_touhou_crisis_faction = yes }
    }
    immediate = {
        fromfrom = {
            solar_system = {
                if = {
                    limit = { NOT = { is_variable_set = lunar_capital_psi_9_count } }
                    set_variable = { which = lunar_capital_psi_9_count value = 0 }
                }
                if = {
                    limit = { NOT = { has_star_flag = lunar_capital_psi_9_fired_on_@root.fromfromfrom } }
                    set_timed_star_flag = { flag = lunar_capital_psi_9_fired_on_@root.fromfromfrom days = 1 }
                    change_variable = { which = lunar_capital_psi_9_count value = 1 }
                }
            }
            if = {
                limit = { NOT = { is_variable_set = lunar_capital_psi_9_count_fleet } }
                set_variable = { which = lunar_capital_psi_9_count_fleet value = 0 }
            }
            if = {
                limit = {
                    solar_system = { is_variable_set = lunar_capital_psi_9_count }
                    check_variable = {
                        which = lunar_capital_psi_9_count_fleet
                        value != solar_system.lunar_capital_psi_9_count
                    }
                }
                set_fleet_flag = touhou_lunar_psi_extra_11
                set_variable = {
                    which = lunar_capital_psi_9_count_fleet
                    value = solar_system.lunar_capital_psi_9_count
                }
                every_owned_ship = {
                    set_variable = { which = lunar_capital_psi_9_count value = prev.lunar_capital_psi_9_count_fleet }
                    remove_modifier = touhou_lunar_psi_extra_11
                    add_modifier = {
                        modifier = touhou_lunar_psi_extra_11
                        mult = lunar_capital_psi_9_count
                    }
                }
            }
        }
    }
}
