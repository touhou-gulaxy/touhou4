                limit = {
                    has_global_flag = lunar_capital_psionic_crisis_prev_stage_1

            else_if = {
                limit = { always = yes }
                touhou_randomize_variable = { which = situation_lunar_capital_psionic_crisis_stage_0_progress scale = 0.075 }
                change_variable = { which = situation_lunar_capital_psionic_crisis_stage_0_progress value = 0.005 }
            }
            # set other countries' progress
            every_playable_country = {
                limit = { country_has_situation = { SITUATION = situation_lunar_capital_psionic_crisis_stage_0 } }
                set_variable = {
                    which = situation_lunar_capital_psionic_crisis_stage_0_progress
                    value = root.owner.situation_lunar_capital_psionic_crisis_stage_0_progress
                }
            }
            # lunar_capital_psionic_crisis_prev_stage_2
            if = {
                limit = { prev = { situation_progress >= 96 } }
                country_event = { id = lunar_capital_psi.203 days = 5 random = 5 }
            }
        }
    }
}
